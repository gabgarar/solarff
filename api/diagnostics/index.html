    Energy, helicity and topology¶

    solarff.helicity ¶

    Energy decomposition and relative magnetic helicity.

    1. Volume potential. \(\mathbf{B}_p = \nabla\phi\) with \(\nabla^2\phi = 0\) in \(V\) and \(\partial\phi/\partial n = \mathbf{B}\cdot\mathbf{n}\) on the SIX faces of the domain. It is the right reference potential for Thomson's theorem (\(E \ge E_p\)) and for the relative helicity. Solved with a discrete cosine transform (DCT-I), exact for the 7-point Laplacian with Neumann conditions.

    2. Energy decomposition (Valori et al. 2013). Each part (potential \(\mathbf{B}_p\) and current-carrying \(\mathbf{B}_J = \mathbf{B} - \mathbf{B}_p\)) is split into solenoidal and non-solenoidal (\(\mathbf{X}_{ns} = \nabla\chi\), \(\nabla^2\chi = \nabla\cdot\mathbf{X}\), \(\partial\chi/\partial n = 0\)):

    \[ \begin{aligned} E &= E_{p,s} + E_{J,s} + E_{p,ns} + E_{J,ns} + E_{\rm mix} \\ E_{\rm div} &= E_{p,ns} + E_{J,ns} + |E_{\rm mix}| \end{aligned} \]

    \(E_{\rm div}/E\) measures how much energy is due to the numerical divergence. If it is large, the energy (and above all the helicity) are not reliable.

    3. Relative helicity (Finn & Antonsen 1985; Berger & Field 1984) in the DeVore gauge (\(A_z = 0\); Valori et al. 2012):

    \[ \begin{aligned} H_V &= \int_V (\mathbf{A} + \mathbf{A}_p)\cdot(\mathbf{B} - \mathbf{B}_p)\,dV \\ H_J &= \int_V (\mathbf{A} - \mathbf{A}_p)\cdot(\mathbf{B} - \mathbf{B}_p)\,dV && \text{(helicity of the current-carrying part)} \\ H_{PJ} &= 2\int_V \mathbf{A}_p\cdot(\mathbf{B} - \mathbf{B}_p)\,dV && \text{(mutual helicity)} \end{aligned} \]

    The index \(|H_J|/|H_V|\) (Pariat et al. 2017) has been proposed as an indicator of the eruptive capacity of a region.

    Units: energy in erg, helicity in Mx² (B in G, lengths in Mm).

    def poisson_neumann(f: np.ndarray, h: float, flux: dict | None = None) -> np.ndarray:
    """Solve ∇²φ = f on a grid of nodes with ∂φ/∂n prescribed on the faces.

    ``flux`` = {"x0": g, "x1": g, "y0": ..., "y1": ..., "z0": ..., "z1": ...}
    with g = ∂φ/∂x (or ∂y, ∂z) on that face (Cartesian component, not outward
    normal). Without ``flux``: homogeneous Neumann. The constant component is
    removed (the solution is defined up to a constant); if the problem
    is not compatible (net flux ≠ ∫f) it is projected onto the closest compatible one.
    """
    rhs = np.array(f, dtype=float, copy=True)
    nx, ny, nz = rhs.shape
    if flux:
        # Ghost points: φ_-1 = φ_1 − 2h g0  →  term −2 g0/h at node 0
        if "x0" in flux:
            rhs[0] += 2 * flux["x0"] / h
        if "x1" in flux:
            rhs[-1] -= 2 * flux["x1"] / h
        if "y0" in flux:
            rhs[:, 0] += 2 * flux["y0"] / h
        if "y1" in flux:
            rhs[:, -1] -= 2 * flux["y1"] / h
        if "z0" in flux:
            rhs[:, :, 0] += 2 * flux["z0"] / h
        if "z1" in flux:
            rhs[:, :, -1] -= 2 * flux["z1"] / h
    F = dctn(rhs, type=1)
    lam = (_eig(nx, h)[:, None, None] + _eig(ny, h)[None, :, None] + _eig(nz, h)[None, None, :])
    lam[0, 0, 0] = 1.0
    F = F / lam
    F[0, 0, 0] = 0.0
    return idctn(F, type=1)

    def volume_potential(field: VectorField3D) -> VectorField3D:
    """Potential field with the same normal component as ``field`` on the six faces."""
    h = field.dx
    bx, by, bz = field.bx, field.by, field.bz
    flux = {"x0": bx[0], "x1": bx[-1], "y0": by[:, 0], "y1": by[:, -1],
            "z0": bz[:, :, 0], "z1": bz[:, :, -1]}
    phi = poisson_neumann(np.zeros(field.shape), h, flux)
    gx, gy, gz = _grad(phi, h)
    # The normal component on the faces is the imposed one (the gradient with one-sided
    # differences does not reproduce it exactly, and that breaks the identity E − E_p = E_J; revision R10)
    gx[0], gx[-1] = bx[0], bx[-1]
    gy[:, 0], gy[:, -1] = by[:, 0], by[:, -1]
    gz[:, :, 0], gz[:, :, -1] = bz[:, :, 0], bz[:, :, -1]
    return VectorField3D(gx, gy, gz, dx=h, origin=field.origin,
                         meta={"method": "potencial (volumen)", "phi": phi})

    def energy_decomposition(field: VectorField3D, potential: VectorField3D | None = None) -> dict:
    """Decomposition of Valori et al. (2013). Returns the energies (erg) and E_div/E."""
    h = field.dx
    if potential is None:
        potential = volume_potential(field)
    B = [field.bx, field.by, field.bz]
    P = [potential.bx, potential.by, potential.bz]
    J = [B[i] - P[i] for i in range(3)]
    Ps, Pns = _split(P, h)
    Js, Jns = _split(J, h)
    E = _e(B, h)
    out = {"E": E, "E_p": _e(P, h), "E_J": _e(J, h),
           "E_p_s": _e(Ps, h), "E_J_s": _e(Js, h), "E_p_ns": _e(Pns, h), "E_J_ns": _e(Jns, h)}
    out["E_mix"] = (_dot(Ps, Jns, h) + _dot(Pns, Js, h) + _dot(Ps, Pns, h) + _dot(Js, Jns, h))
    out["E_div"] = out["E_p_ns"] + out["E_J_ns"] + abs(out["E_mix"])
    out["E_div_frac"] = out["E_div"] / E
    out["E_free"] = E - out["E_p"]
    return out

    def devore_potential(field: VectorField3D, z0: str = "top"):
    """Vector potential in the DeVore gauge (A_z = 0) referred to the ``z0`` face."""
    h = field.dx
    bx, by, bz = field.bx, field.by, field.bz
    k0 = -1 if z0 == "top" else 0
    a0x, a0y = _a0_plane(bz[:, :, k0], h)
    ibx = cumulative_trapezoid(bx, dx=h, axis=2, initial=0)
    iby = cumulative_trapezoid(by, dx=h, axis=2, initial=0)
    if z0 == "top":
        ibx = ibx - ibx[:, :, -1:]
        iby = iby - iby[:, :, -1:]
    # A = A_0 − ẑ × ∫_{z0}^{z} B dz'
    ax = a0x[:, :, None] + iby
    ay = a0y[:, :, None] - ibx
    az = np.zeros_like(ax)
    return ax, ay, az

    def relative_helicity(field: VectorField3D, potential: VectorField3D | None = None,
                      z0: str = "top") -> dict:
    """Relative helicity $H_V$, $H_J$, $H_{PJ}$ (Mx²) and the index $|H_J|/|H_V|$."""
    h = field.dx
    if potential is None:
        potential = volume_potential(field)
    A = devore_potential(field, z0)
    Ap = devore_potential(potential, z0)
    B = [field.bx, field.by, field.bz]
    P = [potential.bx, potential.by, potential.bz]
    D = [B[i] - P[i] for i in range(3)]
    # units: A [G·Mm], B [G], dV [Mm³]  →  ×1e8 (A) ×1e24 (dV) = 1e32 → Mx²
    fac = h ** 3 * MM * MM ** 3
    w = trapezoid_weights(D[0].shape)
    hv = float(sum((w * (A[i] + Ap[i]) * D[i]).sum() for i in range(3)) * fac)
    hj = float(sum((w * (A[i] - Ap[i]) * D[i]).sum() for i in range(3)) * fac)
    hpj = float(2 * sum((w * Ap[i] * D[i]).sum() for i in range(3)) * fac)
    return {"H_V": hv, "H_J": hj, "H_PJ": hpj,
            "ratio_HJ_HV": abs(hj) / abs(hv) if hv else float("nan"), "gauge": f"DeVore ({z0})"}

    def helicity_report(field: VectorField3D) -> dict:
    """Decomposed energy + helicity in two gauges (the difference measures the numerical error)."""
    pot = volume_potential(field)
    e = energy_decomposition(field, pot)
    ht = relative_helicity(field, pot, "top")
    hb = relative_helicity(field, pot, "bottom")
    gauge_diff = abs(ht["H_V"] - hb["H_V"]) / max(abs(ht["H_V"]), abs(hb["H_V"]), 1e-300)
    return {"energy": e, "helicity": ht, "helicity_bottom_gauge": hb, "gauge_diff": gauge_diff,
            "potential": pot}

    solarff.topology ¶

    Field topology: twist number (Tw) and squashing factor (Q).

    Twist number (Berger & Prior 2006; Liu et al. 2016):

    \[ T_w = \frac{1}{4\pi}\int \alpha\,dl, \qquad \alpha = \frac{(\nabla\times\mathbf{B})\cdot\mathbf{B}}{B^2} \]

    Counts how many turns the field makes around a line. \(|T_w| \gtrsim 1\) indicates a flux rope, a structure that is a candidate to erupt.

    Squashing factor (Titov et al. 2002; Pariat & Démoulin 2012): measures how much a small bundle of lines is deformed between its two ends. For the endpoint mapping \((x_0, y_0) \to (X, Y)\) with Jacobian \(\begin{pmatrix} a & b \\ c & d \end{pmatrix}\):

    \[ Q = \frac{a^2 + b^2 + c^2 + d^2}{|B_{n,0} / B_{n,1}|} \]

    \(Q = 2\) is the minimum (no deformation); \(Q \gg 2\) marks the quasi-separatrix layers (QSL), where the connectivity changes abruptly and reconnection is likely.

    def twist_number(field: VectorField3D, lines: list[FieldLine]) -> np.ndarray:
    """Tw of each already traced line (integrates α over its points)."""
    a = _alpha_full(field)
    origin = np.asarray(field.origin)
    tw = np.full(len(lines), np.nan)
    for i, l in enumerate(lines):
        if len(l.points) < 2:
            continue
        mid = 0.5 * (l.points[1:] + l.points[:-1])
        seg = np.linalg.norm(np.diff(l.points, axis=0), axis=1)
        av = map_coordinates(a, ((mid - origin) / field.dx).T, order=1, mode="nearest")
        tw[i] = (av * seg).sum() / (4 * np.pi)
    return tw

    def twist_map(field: VectorField3D, step: int = 1, threshold: float = 0.05,
              ds: float = 0.25) -> dict:
    """Tw of the lines leaving each pixel of the base (as in Liu et al. 2016).

    Each line is integrated from the base in the direction the field leaves
    until it returns to the photosphere or leaves the domain.

    Returns a dict with ``tw`` (2D, NaN where |Bz| is weak), ``closed`` (2D,
    the line returns to the base), ``length`` (Mm) and the coordinates ``x``, ``y``.
    """
    IX, IY, strong = _base_seeds(field, step, threshold)
    bz0 = field.bz[IX, IY, 0]
    start = np.stack([IX[strong], IY[strong], np.zeros(strong.sum())]).astype(float)
    sgn = np.sign(bz0[strong])
    a = _alpha_full(field)
    tw = np.full(IX.shape, np.nan)
    closed = np.zeros(IX.shape, dtype=bool)
    length = np.full(IX.shape, np.nan)
    for s in (1.0, -1.0):
        sel = sgn == s
        if not sel.any():
            continue
        end, L, integ = integrate_lines(field, start[:, sel], s, scalar=a, ds=ds)
        # The sign of the integration does not change α; Tw is intrinsic to the line
        idx = np.flatnonzero(strong.ravel())[sel]
        tw.flat[idx] = integ * field.dx / (4 * np.pi)
        closed.flat[idx] = (end[2] <= 0.5) & (L > 2)
        length.flat[idx] = L * field.dx
    x = field.origin[0] + IX[:, 0] * field.dx
    y = field.origin[1] + IY[0, :] * field.dx
    return {"tw": tw, "closed": closed, "length": length, "x": x, "y": y, "step": step}

    def squashing_q(field: VectorField3D, step: int = 1, threshold: float = 0.05,
                delta: float = 0.25, ds: float = 0.2) -> dict:
    """Squashing factor Q at the base (z = 0).

    For each pixel the central line and four neighbours displaced by
    ±``delta`` cells in x and y are traced; the Jacobian of the endpoint mapping is
    obtained by centred differences. Works for lines ending on
    any face of the domain (Pariat & Démoulin 2012). Where the neighbours
    end on different faces, Q is undefined (NaN).

    Returns ``q`` (2D), ``logq`` (log10 Q), ``face`` (arrival face) and ``x``, ``y``.
    """
    IX, IY, strong = _base_seeds(field, step, threshold)
    upper = np.array(field.shape, dtype=float) - 1
    n = int(strong.sum())
    cx, cy = IX[strong].astype(float), IY[strong].astype(float)
    sgn = np.sign(field.bz[IX[strong], IY[strong], 0])
    offsets = [(0, 0), (delta, 0), (-delta, 0), (0, delta), (0, -delta)]
    ends = np.zeros((5, 3, n))
    starts = np.zeros((5, 3, n))
    for k, (ox, oy) in enumerate(offsets):
        start = np.stack([np.clip(cx + ox, 0, upper[0]), np.clip(cy + oy, 0, upper[1]), np.zeros(n)])
        starts[k] = start
        for s in (1.0, -1.0):
            sel = sgn == s
            if sel.any():
                e, _, _ = integrate_lines(field, start[:, sel], s, ds=ds)
                ends[k][:, sel] = e
    faces = [_face_of(ends[k], upper) for k in range(5)]
    same = np.ones(n, dtype=bool)
    for k in range(1, 5):
        same &= faces[k] == faces[0]
    same &= faces[0] != ""

    q = np.full(n, np.nan)
    fa = (field.bx, field.by, field.bz)
    bn0 = np.abs(map_coordinates(field.bz, np.stack([cx, cy, np.zeros(n)]), order=1, mode="nearest"))
    for name in _FACES:
        sel = same & (faces[0] == name)
        if not sel.any():
            continue
        (u, v), nd = _FACE_DIMS[name]
        # Actual separation of the seeds (at the edges the neighbour is clipped to the face)
        hx = starts[1][0, sel] - starts[2][0, sel]
        hy = starts[3][1, sel] - starts[4][1, sel]
        hx = np.where(hx > 0, hx, np.nan)
        hy = np.where(hy > 0, hy, np.nan)
        a = (ends[1][u, sel] - ends[2][u, sel]) / hx
        b = (ends[3][u, sel] - ends[4][u, sel]) / hy
        c = (ends[1][v, sel] - ends[2][v, sel]) / hx
        d = (ends[3][v, sel] - ends[4][v, sel]) / hy
        bn1 = np.abs(map_coordinates(fa[nd], ends[0][:, sel], order=1, mode="nearest"))
        with np.errstate(divide="ignore", invalid="ignore"):
            q[sel] = (a * a + b * b + c * c + d * d) * bn1 / bn0[sel]
    Q = np.full(IX.shape, np.nan)
    Q[strong] = q
    F = np.full(IX.shape, "", dtype=object)
    F[strong] = faces[0]
    with np.errstate(divide="ignore", invalid="ignore"):
        logq = np.log10(np.where(Q > 0, Q, np.nan))
    x = field.origin[0] + IX[:, 0] * field.dx
    y = field.origin[1] + IY[0, :] * field.dx
    return {"q": Q, "logq": logq, "face": F, "x": x, "y": y, "step": step}

    solarff.stability ¶

    Decay index of the confining field (torus instability).

    A flux rope is held down by the external field passing over it. If that field decays too fast with height, the rope becomes unstable and may erupt (Kliem & Török 2006):

    n(z) = − d ln B_h / d ln z

    where B_h is the horizontal component of the potential field (the "external" field). For a thin torus the theoretical threshold is n_crit = 1.5; in practice values between ~1.1 and ~1.75 are quoted. The critical height h_crit is the height at which n reaches n_crit above the polarity inversion line.

    def pil_mask(bz: np.ndarray, frac: float = 0.15, width: int = 1) -> np.ndarray:
    """Pixels on the polarity inversion line, only between strong fields.

    A pixel is PIL if (smoothed) Bz changes sign with respect to a neighbour and
    there is field > ``frac``·max|Bz| in both polarities within 2 pixels.
    """
    s = gaussian_filter(bz, 1.0)
    sign = np.sign(s)
    m = np.zeros_like(bz, dtype=bool)
    m[:-1, :] |= sign[:-1, :] != sign[1:, :]
    m[:, :-1] |= sign[:, :-1] != sign[:, 1:]
    thr = frac * np.abs(bz).max()
    pos = maximum_filter(np.where(bz > 0, bz, 0), size=5) > thr
    neg = maximum_filter(np.where(bz < 0, -bz, 0), size=5) > thr
    m &= pos & neg
    if width > 1:
        m = maximum_filter(m, size=width)
    return m

    solarff.metrics ¶

    Quality metrics and comparison between models.

    Force-free quality (Wheatland et al. 2000; Schrijver et al. 2006):

    \[ \mathrm{CW}_{\sin} = \frac{\sum_i |\mathbf{J}_i|\,\sigma_i}{\sum_i |\mathbf{J}_i|}, \qquad \sigma_i = \frac{|\mathbf{J}_i\times\mathbf{B}_i|}{|\mathbf{J}_i|\,|\mathbf{B}_i|}, \qquad \langle|f_i|\rangle = \Big\langle\frac{|\nabla\cdot\mathbf{B}|\,\Delta x}{6\,|\mathbf{B}|}\Big\rangle \]

    \(\mathrm{CW}_{\sin}\) is the sine of the J–B angle weighted by the current (0 = perfectly force-free), \(\langle|f_i|\rangle\) the non-solenoidal flux fraction, and \(L_f\), \(L_d\) the terms of Wiegelmann's functional.

    Comparison between two fields \(\mathbf{B}\) (reference) and \(\mathbf{b}\) (Schrijver et al. 2006):

    \[ \begin{aligned} C_{\rm vec} &= \frac{\sum \mathbf{B}\cdot\mathbf{b}}{\sqrt{\sum B^2\,\sum b^2}}, & C_{\rm CS} &= \Big\langle\frac{\mathbf{B}\cdot\mathbf{b}}{|\mathbf{B}|\,|\mathbf{b}|}\Big\rangle, \\ E'_n &= 1 - \frac{\sum |\mathbf{b}-\mathbf{B}|}{\sum |\mathbf{B}|}, & E'_m &= 1 - \Big\langle\frac{|\mathbf{b}-\mathbf{B}|}{|\mathbf{B}|}\Big\rangle, \\ \epsilon &= \frac{\sum |\mathbf{b}|^2}{\sum |\mathbf{B}|^2} && \text{(energy ratio)} \end{aligned} \]

    All equal 1 for identical fields.

    def force_free_metrics(field: VectorField3D, margin: int = 0) -> dict:
    """Force-free and solenoidal quality metrics.

    ``margin``: points to exclude from the side and top edges (e.g. the
    buffer layer ``nd`` of the NLFFF).
    """
    h = field.dx
    jx, jy, jz = field.current()
    d = field.divergence()
    s = _inner(field, margin)
    # Interior points only (the derivatives at the edges are less accurate)
    core = (slice(1, -1), slice(1, -1), slice(1, -1))
    m = np.zeros(field.shape, dtype=bool)
    m[core] = True
    mm = np.zeros(field.shape, dtype=bool)
    mm[s] = True
    m &= mm

    bx, by, bz = field.bx[m], field.by[m], field.bz[m]
    jx, jy, jz, d = jx[m], jy[m], jz[m], d[m]
    b = np.sqrt(bx ** 2 + by ** 2 + bz ** 2)
    j = np.sqrt(jx ** 2 + jy ** 2 + jz ** 2)
    fx, fy, fz = cross(jx, jy, jz, bx, by, bz)
    jxb = np.sqrt(fx ** 2 + fy ** 2 + fz ** 2)
    ok = (b > 0) & (j > 0)
    sigma = np.zeros_like(b)
    sigma[ok] = jxb[ok] / (j[ok] * b[ok])
    cwsin = float((j * sigma).sum() / j.sum()) if j.sum() > 0 else 0.0
    bs = np.where(b > 0, b, np.inf)
    fi = float(np.mean(np.abs(d) * h / (6 * bs)))
    Lf = float(np.mean(np.where(b > 0, jxb ** 2 / bs ** 2, 0)))
    Ld = float(np.mean(d ** 2))
    return {
        "CWsin": cwsin,
        "theta_J_deg": float(np.degrees(np.arcsin(min(cwsin, 1.0)))),
        "<|f_i|>": fi,
        "L_f": Lf,
        "L_d": Ld,
        "E (erg)": energy(field, margin),
    }

    def compare_vector_fields(ref: VectorField3D, b: VectorField3D, margin: int = 0) -> dict:
    """Schrijver et al. (2006) metrics of ``b`` with respect to ``ref``."""
    if ref.shape != b.shape:
        raise ValueError("Los campos deben tener la misma forma")
    s = _inner(ref, margin)
    B = np.stack([ref.bx[s], ref.by[s], ref.bz[s]]).reshape(3, -1)
    bb = np.stack([b.bx[s], b.by[s], b.bz[s]]).reshape(3, -1)
    nB = np.linalg.norm(B, axis=0)
    nb = np.linalg.norm(bb, axis=0)
    ok = (nB > 0) & (nb > 0)
    diff = np.linalg.norm(bb - B, axis=0)
    return {
        "C_vec": float((B * bb).sum() / np.sqrt((nB ** 2).sum() * (nb ** 2).sum())),
        "C_CS": float(np.mean((B * bb).sum(axis=0)[ok] / (nB[ok] * nb[ok]))),
        "E'_n": float(1 - diff.sum() / nB.sum()),
        "E'_m": float(1 - np.mean(diff[ok] / nB[ok])),
        "eps": float((nb ** 2).sum() / (nB ** 2).sum()),
    }

    solarff.montecarlo ¶

    Monte Carlo uncertainties with the HMI measurement errors.

    The full extrapolation (preprocessing + NLFFF) is repeated on n realizations of the magnetogram, each with added Gaussian noise of standard deviation equal to the measurement error of each pixel (*_err segments of SHARP), multiplied by error_scale. The spread of the results gives the uncertainty due to the noise of the data (not that of the method).

    The stability of each field line is measured too: the fraction of realizations in which the line traced from the same seed still matches that of the nominal model (deviation < 10 % of its length and same topology). An unstable line depends on the noise and is not reliable.

    The summary includes bias (Monte Carlo median − nominal) and corrected (2·nominal − median, the bootstrap correction). The validation with exact fields (examples/05_bias_validation.py) shows that this correction helps with isotropic noise but worsens the free energy with HMI-like noise (transverse ≫ longitudinal), so the report does not apply it: the bias is given as a systematic uncertainty.

    Limitations: the formal errors of the HMI inversion underestimate the real uncertainty (they include neither systematics nor errors of the 180° ambiguity) and are treated as independent between pixels and components.

    def perturb(mag: Magnetogram, rng: np.random.Generator, error_scale: float = 1.0,
            noise: float | None = None) -> Magnetogram:
    """One realization: mag + N(0, σ) in each component (σ = pixel error or ``noise``)."""
    def sig(e, b):
        if noise is not None:
            return np.full_like(b, noise)
        if e is None:
            raise ValueError("El magnetograma no tiene mapas de error: pasa noise=<G> o "
                             "descarga los segmentos *_err (download_sharp los incluye por defecto)")
        return e
    bz = mag.bz + rng.normal(0, 1, mag.shape) * sig(mag.bz_err, mag.bz) * error_scale
    bx = mag.bx + rng.normal(0, 1, mag.shape) * sig(mag.bx_err, mag.bx) * error_scale
    by = mag.by + rng.normal(0, 1, mag.shape) * sig(mag.by_err, mag.by) * error_scale
    out = Magnetogram(bz=bz - bz.mean() + mag.bz.mean(), bx=bx, by=by, dx=mag.dx,
                      origin=mag.origin, meta=dict(mag.meta),
                      bx_err=mag.bx_err, by_err=mag.by_err, bz_err=mag.bz_err)
    return out

    def base_seeds(field: VectorField3D, step: int, threshold: float = 0.05) -> np.ndarray:
    """Seeds at the base pixels with |Bz| > ``threshold``·max (as in the confidence map)."""
    bz = field.bz[:, :, 0]

        if checkpoint:
            save_checkpoint(checkpoint, {"signature": signature, "values": values, "good": good, "done": k + 1})
    values = {k: np.array(v, dtype=float) for k, v in values.items()}
    summary = {}
    for k, v in values.items():
        f = v[np.isfinite(v)]
        if f.size:
            p16, p50, p84 = np.percentile(f, [16, 50, 84])
            nom = float(nominal[k]) if np.isfinite(nominal[k]) else None
            d = {"mean": float(f.mean()), "std": float(f.std(ddof=1)) if f.size > 1 else 0.0,
                 "p16": float(p16), "p50": float(p50), "p84": float(p84), "nominal": nom}
            if nom is not None:
                # Bootstrap bias: the added noise shifts the result the way the real
                # noise shifted the nominal value with respect to the noise-free one.
                d["bias"] = float(p50 - nom)
                d["corrected"] = float(2 * nom - p50)
                d["outside"] = bool(nom < p16 or nom > p84)
            summary[k] = d
    stability = good.mean(axis=0)
    say(f"  Monte Carlo terminado: {n} realizaciones en {time.time() - t0:.0f}s")
    return {"values": values, "nominal": nominal, "summary": summary, "stability": stability,
            "seeds": seeds, "lines_nominal": lines_nom, "step": step,
            "config": {"n": n, "error_scale": error_scale, "noise": noise, "seed": seed,
                       "source": "noise" if noise is not None else "HMI *_err",
                       "code_version": code_version()}}

    solarff.budget ¶

    Uncertainty budget of the free energy and the helicity.

    Gathers in one table every source of uncertainty the package knows how to measure, to see at a glance which one dominates:

    • Random noise (this case): half-width of the 16–84 % interval of the Monte Carlo with the HMI error maps, relative to the median.
    • Noise bias (this case): median of the Monte Carlo with respect to the nominal value. It is not corrected (see solarff/bias_validation.json).
    • Method versus the exact solution (synthetic): error of the NLFFF, without noise, with respect to the exact field, on the fields of 05_bias_validation.py.
    • Preprocessing, NLFFF parameters, buffer layer and resolution (synthetic): change of the error with respect to the exact field when each choice is varied around the default value (06_robustness.py, mean of four cases).
    • NLFFF method (this case): spread between the optimisation and Grad–Rubin, if it converged.

    The values are signed relative shifts (lo, hi) (lo ≤ 0 ≤ hi for the ranges). They are not added up: the sources are neither independent nor of the same statistical meaning (one is a standard deviation, others are ranges when a choice is varied), so the total would be misleading. The table serves to identify the dominant source and to quote each one separately.

    def uncertainty_budget(A: dict | None = None, robustness: dict | None = None,
                       bias: dict | None = None) -> dict:
    """Uncertainty budget.

    ``A`` is the analysis of the report (``analysis.analyze``); if missing, only the
    sources measured with synthetic fields are included. ``robustness`` and
    ``bias`` are the summaries of ``06_robustness.py`` and
    ``05_bias_validation.py`` (by default, those shipped with the package).

    Returns ``{"rows": [...], "dominant": {q: key}, "definition": {...}}``.
    Each row: ``key``, ``kind`` (``random``/``systematic``), ``scope``
    (``case``/``synthetic``), ``values`` {q: (lo, hi) or None}, ``detail``.
    """
    A = A or {}
    robustness = robustness if robustness is not None else _load("robustness.json")
    bias = bias if bias is not None else _load("bias_validation.json")
    rows = []

    mc = A.get("mc")
    if mc:
        s = mc["summary"]
        rnd, bia = {}, {}
        for q in QUANTITIES:
            d = s.get(q)
            if d and d.get("p50"):
                h = (d["p84"] - d["p16"]) / 2 / abs(d["p50"])
                rnd[q] = (-h, h)
                b = d["p50"] / d["nominal"] - 1 if d.get("nominal") else None
                bia[q] = (min(0.0, b), max(0.0, b)) if b is not None else None
            else:
                rnd[q] = bia[q] = None
        rows.append({"key": "noise_random", "kind": "random", "scope": "case", "values": rnd,
                     "detail": {"n": mc["n"], "error_scale": mc["error_scale"]}})
        rows.append({"key": "noise_bias", "kind": "systematic", "scope": "case", "values": bia,
                     "detail": {"n": mc["n"]}})

    if bias and bias.get("refs"):
        vals = {}
        for q in QUANTITIES:
            e = [r["model_clean"][q] / r["exact"][q] - 1 for r in bias["refs"].values()
                 if r["exact"].get(q)]
            vals[q] = (min(0.0, min(e)), max(0.0, max(e))) if e else None
        rows.append({"key": "method_exact", "kind": "systematic", "scope": "synthetic", "values": vals,
                     "detail": {"fields": list(bias["refs"]), "size": bias.get("size")}})

    par = (robustness or {}).get("parameters") or {}
    if par:
        for key, prefixes in _GROUPS.items():
            vals, names = {}, set()
            for q in QUANTITIES:
                vals[q], nm = _robustness_range(par, prefixes, q)
                names.update(nm)
            if any(v is not None for v in vals.values()):
                rows.append({"key": key, "kind": "systematic", "scope": "synthetic", "values": vals,
                             "detail": {"variants": sorted(names), "size": robustness.get("size"),
                                        "cases": robustness.get("cases")}})

    M = A.get("methods")
    if M:
        sp = M.get("spread") or {}
        vals = {}
        for q in QUANTITIES:
            d = sp.get(q)
            vals[q] = (-d["rel_halfrange"], d["rel_halfrange"]) if d and d.get("rel_halfrange") is not None else None
        rows.append({"key": "method_gr", "kind": "systematic", "scope": "case", "values": vals,
                     "detail": {"excluded": M.get("excluded") or [],
                                "estimated": any(v is not None for v in vals.values())}})

    dominant = {}
    for q in QUANTITIES:
        best = max(((max(abs(r["values"][q][0]), abs(r["values"][q][1])), r["key"]) for r in rows
                    if r["values"].get(q) is not None), default=None)
        dominant[q] = best[1] if best else None

    definition = None
    nl = (M or {}).get("models", {}).get("nlfff") if M else None
    if nl and nl.get("E_free") and nl.get("E_free_vol"):
        definition = {"E_free": nl["E_free"], "E_free_vol": nl["E_free_vol"],
                      "ratio": nl["E_free_vol"] / nl["E_free"]}
    return {"rows": rows, "dominant": dominant, "definition": definition}
