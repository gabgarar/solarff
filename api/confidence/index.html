class FieldLine:
    """Field line in physical coordinates (Mm).

    ``points`` is ordered in the direction of B: the first point is the
    "outgoing" end (positive polarity if the line is closed) and the last the
    "incoming" end (negative polarity).
    """

    points: np.ndarray       # (M, 3)
    seed: np.ndarray         # (3,)
    closed: bool             # both ends at the base (z ≈ 0)

    @property
    def start(self):
        return self.points[0]

    @property
    def end(self):
        return self.points[-1]

    @property
    def length(self) -> float:
        return float(np.linalg.norm(np.diff(self.points, axis=0), axis=1).sum())

    @property
    def apex(self) -> float:
        return float(self.points[:, 2].max())

    def integrate_lines(field: VectorField3D, start_idx: np.ndarray, sign: float,
                    scalar: np.ndarray | None = None, ds: float = 0.25,
                    max_steps: int = 20000, b_min: float = 1e-6):
    """Integrate lines from ``start_idx`` (3, N) (grid indices) in one direction
    without storing the intermediate points.

    Returns (endpoint (3, N) in indices, length (N,) in cells, integral of
    ``scalar`` along the line (N,) in cell units). Meant for
    thousands of lines (Q and Tw maps).
    """
    fa = (field.bx, field.by, field.bz)
    upper = np.array(field.shape, dtype=float) - 1
    pos = np.clip(np.asarray(start_idx, dtype=float).copy(), 0, upper[:, None])
    n = pos.shape[1]
    length = np.zeros(n)
    integ = np.zeros(n)
    active = np.ones(n, dtype=bool)
    for _ in range(max_steps):
        if not active.any():
            break
        ia = np.flatnonzero(active)
        p = pos[:, ia]
        k1, b = _direction(fa, p, sign)
        k2, _ = _direction(fa, p + 0.5 * ds * k1, sign)
        k3, _ = _direction(fa, p + 0.5 * ds * k2, sign)
        k4, _ = _direction(fa, p + ds * k3, sign)
        newp = p + ds / 6 * (k1 + 2 * k2 + 2 * k3 + k4)
        newp, out = _stop_at_boundary(p, newp, upper)
        stop = out | (b < b_min)
        seg = np.linalg.norm(newp - p, axis=0)
        length[ia] += seg
        if scalar is not None:
            integ[ia] += map_coordinates(scalar, 0.5 * (p + newp), order=1, mode="nearest") * seg
        pos[:, ia] = newp
        active[ia[stop]] = False
    return pos, length, integ

    def trace_field_lines(field: VectorField3D, seeds: np.ndarray, ds: float = 0.5,
                      max_steps: int = 4000, b_min: float = 1e-3) -> list[FieldLine]:
    """Trace field lines from ``seeds`` (array (N, 3) in Mm) in both directions.

    ``ds`` is the step in cell units. Using the *same* seeds in
    different models allows their connectivity to be compared.
    """
    seeds = np.atleast_2d(np.asarray(seeds, dtype=float))
    origin = np.asarray(field.origin, dtype=float)
    idx0 = ((seeds - origin) / field.dx).T  # (3, N)
    upper = np.array(field.shape, dtype=float) - 1
    idx0 = np.clip(idx0, 0, upper[:, None])
    fa = (field.bx, field.by, field.bz)

    fwd = _trace_one_direction(fa, idx0, +1.0, ds, max_steps, upper, b_min)
    bwd = _trace_one_direction(fa, idx0, -1.0, ds, max_steps, upper, b_min)

    lines = []
    z_tol = 1.0  # cells
    for j in range(seeds.shape[0]):
        f = fwd[:, :, j]
        b = bwd[:, :, j]
        f = f[~np.isnan(f[:, 0])]
        b = b[~np.isnan(b[:, 0])]
        pts = np.concatenate([b[::-1], f[1:]], axis=0)
        pts_phys = pts * field.dx + origin
        closed = bool(pts[0, 2] <= z_tol and pts[-1, 2] <= z_tol and len(pts) > 3)
        lines.append(FieldLine(points=pts_phys, seed=seeds[j], closed=closed))
    return lines

    def seeds_from_magnetogram(source: Magnetogram | VectorField3D, n: int = 80,
                           threshold: float = 0.1, z: float | None = None,
                           polarity: str = "positive", method: str = "grid",
                           rng: int | None = 0) -> np.ndarray:
    """Seeds to trace lines at pixels with |Bz| > ``threshold``·max|Bz|.

    **Parameters**

    - ``n``: approximate number of seeds.
    - ``polarity``: ``"positive"`` (default), ``"negative"`` or ``"both"``. Since each
        line is traced in both directions, seeding on a single polarity avoids
        drawing the same loop twice.
    - ``method``: ``"grid"`` (regular grid over the selected region: also covers
        the weak field and gives cleaner figures) or ``"random"``
        (probability proportional to |Bz|, concentrates in the cores).
    - ``z``: height of the seeds (by default one cell above the base, to
        avoid the z = 0 layer, where the NLFFF is noisier).
    """
    if isinstance(source, VectorField3D):
        bz = source.bz[:, :, 0]
        dx = source.dx
        ox, oy, _ = source.origin
    else:
        bz, dx = source.bz, source.dx
        ox, oy = source.origin
    if z is None:
        z = dx
    signed = {"positive": bz, "negative": -bz, "both": np.abs(bz)}[polarity]
    mask = signed > threshold * np.abs(bz).max()
    gen = np.random.default_rng(rng)

    if method == "grid":
        ix, iy = np.nonzero(mask)
        # Grid step such that ~n seeds remain
        step = max(1.0, np.sqrt(mask.sum() / max(n, 1)))
        while True:
            gx = np.floor(ix / step)
            gy = np.floor(iy / step)
            key = gx * 1e6 + gy
            _, first = np.unique(key, return_index=True)
            if first.size <= n or step > max(bz.shape):
                break
            step *= 1.05
        # One point per grid cell: the one closest to the centre of the cell
        cx, cy = (gx + 0.5) * step, (gy + 0.5) * step
        d = (ix - cx) ** 2 + (iy - cy) ** 2
        order = np.lexsort((d, key))
        _, first = np.unique(key[order], return_index=True)
        sel = order[first]
        px, py = ix[sel].astype(float), iy[sel].astype(float)
    elif method == "random":
        cand = np.flatnonzero(mask.ravel())
        p = np.abs(bz).ravel()[cand]
        pick = gen.choice(cand, size=min(n, cand.size), replace=False, p=p / p.sum())
        px, py = np.unravel_index(pick, bz.shape)
        px = px + gen.uniform(-0.5, 0.5, px.size)
        py = py + gen.uniform(-0.5, 0.5, py.size)
    else:
        raise ValueError("method debe ser 'grid' o 'random'")
    return np.column_stack([ox + px * dx, oy + py * dx, np.full(px.size, z)])

    solarff.confidence ¶

    Reliability of each field line of a model.

    For each traced line, indicators are computed that, without knowing the true solution, flag whether that part of the model can be trusted:

    Indicator Meaning
    sin_jb mean sine of the J–B angle along the line, weighted by the current (0 = force-free)
    alpha_spread relative variation of α along the line (in a force-free field α is constant on each line)
    div mean relative divergence \(\langle \lvert\nabla\cdot\mathbf{B}\rvert\,\Delta x / 6\lvert\mathbf{B}\rvert \rangle\) over the line
    buffer fraction of the line inside the buffer layer, where the (unobserved) boundary rules
    open 1 if the line leaves through a side or the top of the domain
    foot_snr observed transverse field at the weakest footpoint divided by its noise (the α carried by the line is measured at the footpoint)
    base_change relative correction of the transverse field that the model applied at the footpoints (relaxed base): how much it ignored the data there
    apex maximum height of the line / height of the box (high lines depend more on the unobserved top)
    length length of the line / horizontal diagonal of the box
    length_h length of the line / height of the box (a line that is long compared with the height depends on the unobserved top, whatever the horizontal size of the box)

    The indicators are combined into a confidence ∈ [0, 1] (1 − probability that the line is wrong) by a logistic regression calibrated with exact force-free fields with noise, where the true error of each line is known. The default calibration (DEFAULT_CALIBRATION) was fitted with the diverse suite of solarff.calibration (6 field families, 3 noise models, 3 resolutions and boxes of four proportions) and validated leaving out each family.

    Limitation: these are necessary conditions, not sufficient ones. A line can be perfectly force-free and still be wrong if the data at its footpoint are consistently erroneous; that is only detected by noise ensembles, independent methods or coronal observations.

    def line_indicators(field: VectorField3D, lines: list[FieldLine], mag: Magnetogram | None = None,
                    noise_bt: float = 100.0, nd: int | None = None) -> dict:
    """Reliability indicators of each line (see the table of the module).

    **Parameters**

    - ``field``: 3D model (normally the NLFFF).
    - ``lines``: lines traced in ``field``.
    - ``mag``: magnetogram *used as boundary condition* (the preprocessed one if
        there was one). Needed for ``foot_snr`` and ``base_change``.
    - ``noise_bt``: noise of the transverse field in G (~100 G in HMI).
    - ``nd``: buffer layer (by default that of the model, ``meta["nd"]``).
    """
    h = field.dx
    origin = np.asarray(field.origin)
    jx, jy, jz = field.current()
    bx, by, bz = field.bx, field.by, field.bz
    fx, fy, fz = cross(jx, jy, jz, bx, by, bz)
    jxb = np.sqrt(fx ** 2 + fy ** 2 + fz ** 2)
    babs = field.babs
    jb = np.sqrt(jx ** 2 + jy ** 2 + jz ** 2) * babs
    alpha = np.nan_to_num(field.alpha(b_min_frac=0.0), nan=0.0, posinf=0.0, neginf=0.0)
    div = np.abs(field.divergence()) * h / (6 * np.maximum(babs, 1e-12 * babs.max()))
    nd = nd if nd is not None else field.meta.get("nd", max(4, min(field.shape) // 10))
    w = weight_function(field.shape, nd)
    upper = np.array(field.shape) - 1
    # reference α for normalisation (scale of the domain)
    alpha0 = 2 * np.pi / (max(field.shape[:2]) * h) * 0.1

    out = {k: np.full(len(lines), np.nan) for k in ALL_FEATURES + ("length_h",)}
    ztop = max(upper[2] * h, 1e-9)
    diag = np.hypot(*(upper[:2] * h))
    for i, l in enumerate(lines):
        p = l.points
        if len(p) < 3:
            continue
        idx = ((p - origin) / h).T
        seg = np.r_[np.linalg.norm(np.diff(p, axis=0), axis=1), 0.0]
        if seg.sum() <= 0:
            continue
        a_jxb = _interp(jxb, idx)
        a_jb = _interp(jb, idx)
        den = (a_jb * seg).sum()
        out["sin_jb"][i] = (a_jxb * seg).sum() / den if den > 0 else 0.0
        al = _interp(alpha, idx)
        wt = seg / seg.sum()
        m = (al * wt).sum()
        sd = np.sqrt(((al - m) ** 2 * wt).sum())
        out["alpha_spread"][i] = sd / (abs(m) + alpha0)
        out["div"][i] = (_interp(div, idx) * wt).sum()
        ww = _interp(w, idx)
        out["buffer"][i] = (wt * (ww < 0.99)).sum()
        ends = np.array([idx[:, 0], idx[:, -1]])
        on_side = ((ends[:, 0] <= 1e-6) | (ends[:, 0] >= upper[0] - 1e-6) |
                   (ends[:, 1] <= 1e-6) | (ends[:, 1] >= upper[1] - 1e-6) |
                   (ends[:, 2] >= upper[2] - 1e-6))
        out["open"][i] = float(on_side.any() or not l.closed)
        out["length"][i] = seg.sum() / diag
        out["length_h"][i] = seg.sum() / ztop
        out["apex"][i] = (p[:, 2].max() - origin[2]) / ztop

    if mag is not None and mag.has_vector:
        # Footpoints: endpoints at the base (z < 1 cell)
        bt_obs = np.hypot(mag.bx, mag.by)
        dbt = np.hypot(field.bx[:, :, 0] - mag.bx, field.by[:, :, 0] - mag.by)
        for i, l in enumerate(lines):
            if len(l.points) < 3:
                continue
            feet = [q for q in (l.points[0], l.points[-1]) if q[2] - origin[2] < h]
            if not feet:
                feet = [l.points[0]]
            fi = np.array(feet)
            gx = (fi[:, 0] - origin[0]) / h
            gy = (fi[:, 1] - origin[1]) / h
            bt = _interp2(bt_obs, gx, gy)
            out["foot_snr"][i] = float(bt.min() / noise_bt)
            out["base_change"][i] = float((_interp2(dbt, gx, gy) / np.maximum(bt, noise_bt)).max())
    else:
        out["foot_snr"][:] = 1.0
        out["base_change"][:] = 0.0
    return out

    def line_errors(lines_model: list[FieldLine], lines_ref: list[FieldLine], n: int = 40) -> dict:
    """True error of each line with respect to the reference traced from the same seed.

    ``shape``: mean distance between the two lines sampled at the same
    fractions of length, relative to the length of the reference line.
    ``endpoint``: mean distance between endpoints (Mm).
    ``topology``: 1 if one is closed and the other open.
    ``shape_abs``: the same mean distance in Mm (not divided by the length);
    ``length_ref`` / ``length_model``: lengths (Mm). They allow "wrong line" to be
    redefined with an absolute threshold without tracing again.
    """
    shape = np.full(len(lines_ref), np.nan)
    shape_abs = np.full(len(lines_ref), np.nan)
    lref = np.full(len(lines_ref), np.nan)
    lmod = np.full(len(lines_ref), np.nan)
    endp = np.full(len(lines_ref), np.nan)
    topo = np.zeros(len(lines_ref))
    for i, (a, b) in enumerate(zip(lines_model, lines_ref)):
        if len(a.points) < 3 or len(b.points) < 3:
            continue
        def res(p):
            s = np.r_[0, np.cumsum(np.linalg.norm(np.diff(p, axis=0), axis=1))]
            t = np.linspace(0, s[-1], n)
            return np.stack([np.interp(t, s, p[:, k]) for k in range(3)], axis=1), s[-1]
        pa, la = res(a.points)
        pb, lb = res(b.points)
        shape_abs[i] = np.linalg.norm(pa - pb, axis=1).mean()
        shape[i] = shape_abs[i] / max(lb, 1e-9)
        lref[i], lmod[i] = lb, la
        endp[i] = 0.5 * (np.linalg.norm(a.start - b.start) + np.linalg.norm(a.end - b.end))
        topo[i] = float(a.closed != b.closed)
    return {"shape": shape, "endpoint": endp, "topology": topo, "shape_abs": shape_abs,
            "length_ref": lref, "length_model": lmod}

    def auc(score, bad):
    """Area under the ROC curve: probability that a wrong line has a higher ``score`` than a good one."""
    score, bad = np.asarray(score, float), np.asarray(bad, bool)
    ok = np.isfinite(score)
    s, b = score[ok], bad[ok]
    if b.all() or (~b).all():
        return float("nan")
    order = np.argsort(s)
    ranks = np.empty(len(s))
    ranks[order] = np.arange(1, len(s) + 1)
    # ties: mean rank
    _, inv, cnt = np.unique(s, return_inverse=True, return_counts=True)
    sums = np.bincount(inv, ranks)
    ranks = (sums / cnt)[inv]
    n1 = b.sum()
    return float((ranks[b].sum() - n1 * (n1 + 1) / 2) / (n1 * (len(s) - n1)))

    solarff.calibration ¶

    Rigorous calibration of the per-line confidence with diverse data.

    Generates a suite of cases with known truth (several families of exact force-free fields, several noise levels and types and several resolutions; see solarff.analytic.CALIBRATION_FIELDS and NOISE_MODELS), and evaluates classifiers with cross-validation leaving out a whole field type (the classifier has never seen the geometry it is evaluated on), with bootstrap confidence intervals.

    Metrics (see Niculescu-Mizil & Caruana 2005; Guo et al. 2017):

    • AUC: ranking ability (do the wrong lines get less confidence?).
    • Brier: mean squared error of the predicted probability.
    • ECE (expected calibration error): mean difference between the predicted probability and the observed frequency of wrong lines.

    The comparison of classifiers uses scikit-learn (optional dependency, pip install scikit-learn); the rest does not need it.

    def bootstrap_ci(p, y, metric, groups=None, n_boot: int = 500, seed: int = 0, level: float = 0.95):
    """Bootstrap confidence interval of ``metric(p, y)``.

    If ``groups`` are given (e.g. the case each line belongs to), whole
    groups are resampled: the lines of the same case are not
    independent, and resampling single lines would give intervals that are too
    narrow.
    """
    rng = np.random.default_rng(seed)
    p, y = np.asarray(p), np.asarray(y)
    vals = []
    if groups is None:
        for _ in range(n_boot):
            i = rng.integers(0, len(p), len(p))
            vals.append(metric(p[i], y[i]))
    else:
        ug = np.unique(groups)
        members = {g: np.flatnonzero(groups == g) for g in ug}
        for _ in range(n_boot):
            i = np.concatenate([members[g] for g in rng.choice(ug, len(ug))])
            vals.append(metric(p[i], y[i]))
    vals = np.asarray(vals, float)
    vals = vals[np.isfinite(vals)]
    a = (1 - level) / 2
    return float(np.quantile(vals, a)), float(np.quantile(vals, 1 - a))

    def cross_validate(data: list, features=USED_FEATURES, models=None, group: str = "field",
                   n_boot: int = 500, verbose: bool = True, l2: float = 1e-2) -> dict:
    """Cross-validation leaving out one value of ``group`` ("field", "noise_model" or "size").

    For each model returns the out-of-sample probabilities of all the
    lines, global AUC/Brier/ECE with bootstrap 95 % CI (resampling
    whole cases) and the metrics of each left-out group. ``l2`` is the penalty
    of the logistic regression (1e-2 in the published model; 0 = maximum likelihood).
    """
    say = print if verbose else (lambda *a, **k: None)
    features = list(features)
    X, y, G = stack(data, features)
    case_id = np.concatenate([[i] * len(d["bad"]) for i, d in enumerate(data)])
    case_id = case_id[np.all(np.isfinite(np.vstack([_design(d["ind"], features) for d in data])), axis=1)]
    groups = G[group]
    cands = classifiers(features)
    cands["logistic"] = (lambda: _Logistic(l2=l2), False)
    models = list(models or cands)
    out = {"features": features, "group": group, "y": y, "groups": groups, "case": case_id,
           "base_rate": float(y.mean()), "models": {}}
    for name in models:
        factory, needs_groups = cands[name]
        p = np.full(len(y), np.nan)
        per = {}
        for g in np.unique(groups):
            te = groups == g
            m = factory()
            if needs_groups:
                inner = G["field"][~te] if group != "field" else groups[~te]
                m.fit(X[~te], y[~te], groups=inner)
            else:
                m.fit(X[~te], y[~te])
            p[te] = m.predict_proba(X[te])[:, 1]
            per[str(g)] = {"auc": auc(p[te], y[te]), "brier": brier(p[te], y[te]),
                           "ece": ece(p[te], y[te]), "bad_rate": float(y[te].mean()), "n": int(te.sum())}
        res = {"p": p, "per_group": per}
        for mname, fn in (("auc", auc), ("brier", brier), ("ece", ece)):
            res[mname] = float(fn(p, y))
            res[mname + "_ci"] = bootstrap_ci(p, y, fn, groups=case_id, n_boot=n_boot)
        aucs = [v["auc"] for v in per.values()]
        res["auc_groups_mean"], res["auc_groups_std"] = float(np.mean(aucs)), float(np.std(aucs))
        res["reliability"] = reliability_curve(p, y)
        out["models"][name] = res
        say(f"  {name:18s} AUC {res['auc']:.3f} [{res['auc_ci'][0]:.3f}, {res['auc_ci'][1]:.3f}]  "
            f"Brier {res['brier']:.3f}  ECE {res['ece']:.3f}  "
            f"(AUC por {group}: {res['auc_groups_mean']:.3f} ± {res['auc_groups_std']:.3f})")
    return out

    def fit_default(data: list, cv_field: dict, features=USED_FEATURES, isotonic: bool = False,
                threshold: float = 0.8) -> dict:
    """Default calibration fitted with the *whole* suite, and its out-of-sample metrics.

    The logistic regression is fitted with all the lines. The metrics, the
    isotonic recalibration (if ``isotonic``) and the threshold are obtained from the
    predictions of the leave-one-field-out cross-validation
    (``cv_field = cross_validate(data, group="field")``), that is, always
    on geometries the model has not seen.
    ``threshold``: minimum confidence of a "reliable" line. Since the probability
    is calibrated (ECE ≈ 0.02 out of sample), it is read literally: 0.8 =
    at most a 20 % probability that the line is wrong.
    Returns ``{"calibration": ..., "metrics": ...}`` with the same format as
    ``solarff.confidence.fit_calibration``.
    """
    features = list(features)
    X, y, _ = stack(data, features)
    cal = _fit_logistic(X, y.astype(float))
    cal.update(features=features, bad_threshold=data[0]["bad_threshold"], noise="varios")
    p_oof = np.asarray(cv_field["models"]["logistic"]["p"], float)
    if isotonic:
        from sklearn.isotonic import IsotonicRegression
        iso = IsotonicRegression(out_of_bounds="clip", y_min=0, y_max=1).fit(p_oof, y)
        grid = np.linspace(0, 1, 101)
        cal["iso_x"], cal["iso_y"] = grid.round(4).tolist(), np.round(iso.predict(grid), 4).tolist()
        p_oof = np.interp(p_oof, cal["iso_x"], cal["iso_y"])
    thr = float(threshold)
    cal["threshold"] = thr
    conf = 1 - p_oof
    acc = conf >= thr
    shape = np.concatenate([d["err"]["shape"] for d in data])
    ok = np.all(np.isfinite(np.vstack([_design(d["ind"], features) for d in data])), axis=1)
    shape = shape[ok]
    metrics = {
        "auc_combined": auc(p_oof, y), "auc_features": single_feature_auc(data),
        "threshold": thr, "bad_fraction_all": float(y.mean()),
        "bad_fraction_accepted": float(y[acc].mean()) if acc.any() else float("nan"),
        "accepted_fraction": float(acc.mean()),
        "error_median_accepted": float(np.nanmedian(shape[acc])) if acc.any() else float("nan"),
        "error_median_rejected": float(np.nanmedian(shape[~acc])) if (~acc).any() else float("nan"),
        "brier": brier(p_oof, y), "ece": ece(p_oof, y),
    }
    groups = np.asarray(cv_field["groups"])
    per = {}
    for g in np.unique(groups):
        m = groups == g
        per[str(g)] = {"accepted": float(acc[m].mean()),
                       "bad_accepted": float(y[m][acc[m]].mean()) if acc[m].any() else float("nan"),
                       "bad_all": float(y[m].mean())}
    metrics["per_field"] = per
    metrics["bad_accepted_worst_field"] = float(np.nanmax([v["bad_accepted"] for v in per.values()]))
    return {"calibration": cal, "metrics": metrics}
