    solarff.analytic ¶

    Analytical NLFFF solution of Low & Lou (1990) to validate the methods.

    Self-similar axisymmetric field in spherical coordinates \((r, \theta, \varphi)\):

    \[ \begin{aligned} A &= \frac{P(\mu)}{r^n}, \qquad \mu = \cos\theta, \qquad Q = \frac{a\,P^{1+1/n}}{r^{n+1}} \\ B_r &= -\frac{P'(\mu)}{r^{n+2}}, \qquad B_\theta = \frac{n\,P}{r^{n+2}\sin\theta}, \qquad B_\varphi = \frac{a\,P^{1+1/n}}{r^{n+2}\sin\theta} \end{aligned} \]

    with

    \[ (1-\mu^2)\,P'' + n(n+1)\,P + a^2\,\frac{1+n}{n}\,P^{1+2/n} = 0, \qquad P(\pm1) = 0, \quad P'(-1) = 10 . \]

    The eigenvalue \(a^2\) is obtained by shooting (\(m\)-th non-trivial root).

    The source is placed at depth l below the photosphere with the axis tilted by an angle phi. Standard case (Schrijver et al. 2006, case I): \(n = 1\), \(m = 1\), \(l = 0.3\), \(\Phi = \pi/4\).

def low_lou_eigenvalue(n: int = 1, m: int = 1) -> float:
    """m-th (non-trivial) eigenvalue a² of the Low & Lou problem."""
    grid = np.linspace(0.01, 10.0, 400)
    vals = np.array([_shoot(n, a2).y[0, -1] for a2 in grid])
    roots = []
    for i in range(len(grid) - 1):
        if np.sign(vals[i]) != np.sign(vals[i + 1]):
            roots.append(brentq(lambda a: _shoot(n, a).y[0, -1], grid[i], grid[i + 1], xtol=1e-12))
            if len(roots) == m:
                return float(roots[-1])
    raise RuntimeError(f"No se encontró el autovalor m={m} para n={n}")

    def low_lou_field(nx: int = 64, ny: int = 64, nz: int = 64, n: int = 1, m: int = 1,
                  l: float = 0.3, phi: float = np.pi / 4, length_mm: float = 50.0,
                  bmax: float = 1000.0) -> VectorField3D:
    """Low & Lou field in the box x, y ∈ [-1, 1], z ∈ [0, (nz-1)·h] (units of ``length_mm``).

    The result is scaled so that max|Bz(z=0)| = ``bmax`` G and the box is
    2·``length_mm`` Mm wide.
    """
    a2 = low_lou_eigenvalue(n, m)
    a = np.sqrt(a2)
    sol = _shoot(n, a2).sol

    h = 2.0 / (nx - 1)
    x = -1 + h * np.arange(nx)
    y = -1 + h * np.arange(ny)  # same step; if ny != nx the box is not square
    y = y - (y[-1] + y[0]) / 2
    z = h * np.arange(nz)
    X0, Y0, Z0 = np.meshgrid(x, y, z, indexing="ij")

    # Coordinates in the frame of the source (rotated by Φ about y, shifted by l)
    zs = Z0 + l
    X = X0 * np.cos(phi) - zs * np.sin(phi)
    Y = Y0
    Z = X0 * np.sin(phi) + zs * np.cos(phi)
    r = np.sqrt(X ** 2 + Y ** 2 + Z ** 2)
    rho = np.maximum(np.hypot(X, Y), 1e-12)
    mu = np.clip(Z / r, -1 + _EPS, 1 - _EPS)
    sint = np.sqrt(1 - mu ** 2)
    cphi, sphi = X / rho, Y / rho

    P, dP = sol(mu.ravel())
    P = P.reshape(mu.shape)
    dP = dP.reshape(mu.shape)
    rn2 = r ** (n + 2)
    br = -dP / rn2
    bt = n * P / (rn2 * sint)
    bp = a * _spow(P, 1 + 1 / n) / (rn2 * sint)

    BX = br * sint * cphi + bt * mu * cphi - bp * sphi
    BY = br * sint * sphi + bt * mu * sphi + bp * cphi
    BZ = br * mu - bt * sint

    bx = BX * np.cos(phi) + BZ * np.sin(phi)
    by = BY
    bz = -BX * np.sin(phi) + BZ * np.cos(phi)

    s = bmax / np.abs(bz[:, :, 0]).max()
    return VectorField3D(bx * s, by * s, bz * s, dx=h * length_mm,
                         origin=(x[0] * length_mm, y[0] * length_mm, 0.0),
                         meta={"method": "Low&Lou", "n": n, "m": m, "a2": a2, "l": l, "phi": phi})

    def low_lou_magnetogram(noise: float = 0.0, rng: int | None = 0, **kw) -> tuple[Magnetogram, VectorField3D]:
    """Return (z=0 magnetogram, exact 3D field) of Low & Lou.

    ``noise``: standard deviation (G) of Gaussian noise added to the three
    components of the magnetogram (to test the preprocessing).
    """
    f = low_lou_field(**kw)
    bx, by, bz = (c.copy() for c in f.bottom())
    if noise > 0:
        g = np.random.default_rng(rng)
        bx += g.normal(0, noise, bx.shape)
        by += g.normal(0, noise, by.shape)
        bz += g.normal(0, noise, bz.shape)
    mag = Magnetogram(bz=bz, bx=bx, by=by, dx=f.dx, origin=f.origin[:2],
                      meta={"source": "Low&Lou", "noise": noise, **f.meta})
    return mag, f

    def gold_hoyle_field(nx: int = 48, ny: int = 48, nz: int = 48, half_width: float = 20.0,
                     b: float = 0.3, b0: float = 1000.0) -> VectorField3D:
    r"""Gold & Hoyle (1960) flux tube, vertical and centred in the box.

    $$
    B_z = \frac{B_0}{1 + b^2r^2}, \qquad B_\varphi = \frac{B_0\,b\,r}{1 + b^2r^2}
    $$

    It is force-free with $\alpha = 2b/(1 + b^2r^2)$ and every line is a helix of
    constant radius. The twist number of a line of radius $r$ between $z = 0$
    and $z = H$ is exactly $T_w = b\,H / \big(2\pi\sqrt{1 + b^2r^2}\big)$
    (see ``gold_hoyle_twist``).
    """
    h, x, y, z = _box(nx, ny, nz, half_width)
    X, Y, _ = np.meshgrid(x, y, z, indexing="ij")
    r2 = X ** 2 + Y ** 2
    den = 1 + b * b * r2
    bz = b0 / den
    bphi_over_r = b0 * b / den
    bx = -bphi_over_r * Y
    by = bphi_over_r * X
    return VectorField3D(bx, by, bz, dx=h, origin=(x[0], y[0], 0.0),
                         meta={"method": "Gold-Hoyle", "b": b, "b0": b0})

    def dipole_field(nx: int = 64, ny: int = 64, nz: int = 40, half_width: float = 50.0,
                 depth: float = 10.0, b0: float = 1000.0) -> VectorField3D:
    """Potential field of a horizontal dipole (moment along +x) buried at
    depth ``depth`` below the centre of the box.

    At the photosphere it gives a bipolar region with the PIL over x = 0. Right above
    the dipole the field is horizontal and decays as $(z + d)^{-3}$, so the
    decay index is exactly $n(z) = 3z / (z + d)$ and reaches the
    torus-instability threshold $n = 1.5$ at $z = d$.
    """
    h, x, y, z = _box(nx, ny, nz, half_width)
    X, Y, Z = np.meshgrid(x, y, z, indexing="ij")
    Zs = Z + depth
    r2 = X ** 2 + Y ** 2 + Zs ** 2
    r5 = r2 ** 2.5
    # B = (3 (m·r) r - m r²) / r^5 with m = x̂
    bx = (3 * X * X - r2) / r5
    by = 3 * X * Y / r5
    bz = 3 * X * Zs / r5
    s = b0 / np.abs(bz[:, :, 0]).max()
    return VectorField3D(bx * s, by * s, bz * s, dx=h, origin=(x[0], y[0], 0.0),
                         meta={"method": "dipolo", "depth": depth})

    def hyperbolic_field(nx: int = 48, ny: int = 48, nz: int = 24, half_width: float = 20.0,
                     lam: float = 0.05, b0: float = 1.0) -> VectorField3D:
    r"""Hyperbolic field $\mathbf{B} = (\lambda x, -\lambda y, B_0)$: potential and divergence-free.

    The lines satisfy $x = x_0\,e^{\lambda z/B_0}$, $y = y_0\,e^{-\lambda z/B_0}$. The mapping between
    the base and the top ($z = H$) has Jacobian $\mathrm{diag}(e^a, e^{-a})$, $a = \lambda H/B_0$, so
    the squashing factor is exactly $Q = 2\cosh(2a)$ over the whole
    domain (see ``hyperbolic_q``). This is the structure of a "hyperbolic
    flux tube" (Titov et al. 2002).
    """
    h, x, y, z = _box(nx, ny, nz, half_width)
    X, Y, Z = np.meshgrid(x, y, z, indexing="ij")
    return VectorField3D(lam * X, -lam * Y, np.full_like(X, b0), dx=h,
                         origin=(x[0], y[0], 0.0), meta={"method": "hiperbólico", "lam": lam})

    def lfff_multipole_field(nx: int = 40, ny: int = 40, nz: int = 28, spots=None,
                         alpha_frac: float = 0.7, length_mm: float = 50.0,
                         bmax: float = 1000.0) -> VectorField3D:
    """*Exact* linear force-free field over a multipolar region.

    ``spots``: list of Gaussian spots ``(x, y, sign·amplitude, width)`` in
    box units (x, y ∈ [-1, 1]). The Fourier solution (Alissandrakis
    1981) is an exact solution of ∇×B = αB in the periodic domain, so
    it serves as a known truth different from Low & Lou: currents distributed
    over the whole region instead of concentrated, and another geometry.
    ``alpha_frac``: α as a fraction of α_max = 2π/L.
    """
    from .lfff import alpha_max, lfff
    if spots is None:  # tilted bipole
        spots = [(-0.35, -0.15, 1.0, 0.18), (0.35, 0.15, -1.0, 0.18)]
    h = 2.0 / (nx - 1)
    x = -1 + h * np.arange(nx)
    y = -(ny - 1) * h / 2 + h * np.arange(ny)
    X, Y = np.meshgrid(x, y, indexing="ij")
    bz = np.zeros_like(X)
    for (x0, y0, amp, w) in spots:
        bz += amp * np.exp(-((X - x0) ** 2 + (Y - y0) ** 2) / (2 * w * w))
    bz -= bz.mean()  # balanced flux (the k = 0 mode is not force-free with α ≠ 0)
    bz *= bmax / np.abs(bz).max()
    mag = Magnetogram(bz=bz, dx=h * length_mm, origin=(x[0] * length_mm, y[0] * length_mm))
    a = alpha_frac * alpha_max(mag)
    f = lfff(mag, a, nz)
    f.meta.update({"method": "LFFF exacto", "alpha_frac": alpha_frac, "spots": spots})
    return f

    titov_demoulin_field ¶

    titov_demoulin_field(nx: int = 40, ny: int = 40, nz: int = 28, R: float = 2.2, a: float = 0.7, d: float = 1.0, L: float = 2.0, n_turns: float = 1.0, handedness: int = -1, half_width: float = 3.0, box_mm: float = 100.0, bmax: float = 1000.0, current_factor: float = 1.0) -> VectorField3D

    Titov & Démoulin (1999) flux rope: current torus in equilibrium.

    B = B_I + B_θ + B_q (Eqs. 16–31 of TD99, with μ0 = 1):

    • B_I: ring of major radius R and minor radius a with uniform toroidal current I, in the plane x = 0 and with its centre at depth d;
    • B_q: charges ±q at (±L, 0, −d), which hold the ring;
    • B_θ: toroidal field of a line current I0 along the symmetry axis (x axis at z = −d), which gives the twist.

    I comes from the force balance (Eq. 6, l_i = 1/2). I0 is set by the number of turns n_turns of the lines in the coronal part of the tube (Eqs. 7–9); handedness = −1 gives left-handed twist, as in Valori et al. (2010). The default parameters are those of Wiegelmann et al. (2006) in units of d (R = 2.2, a = 0.7, d = 1, L = 2).

    def titov_demoulin_field(nx: int = 40, ny: int = 40, nz: int = 28, R: float = 2.2, a: float = 0.7,
                         d: float = 1.0, L: float = 2.0, n_turns: float = 1.0, handedness: int = -1,
                         half_width: float = 3.0, box_mm: float = 100.0,
                         bmax: float = 1000.0, current_factor: float = 1.0) -> VectorField3D:
    """Titov & Démoulin (1999) flux rope: current torus in equilibrium.

    B = B_I + B_θ + B_q (Eqs. 16–31 of TD99, with μ0 = 1):

    * B_I: ring of major radius ``R`` and minor radius ``a`` with uniform toroidal
      current I, in the plane x = 0 and with its centre at depth ``d``;
    * B_q: charges ±q at (±``L``, 0, −``d``), which hold the ring;
    * B_θ: toroidal field of a line current I0 along the symmetry axis
      (x axis at z = −d), which gives the twist.

    I comes from the force balance (Eq. 6, l_i = 1/2). I0 is set by the
    number of turns ``n_turns`` of the lines in the coronal part of the tube
    (Eqs. 7–9); ``handedness`` = −1 gives left-handed twist, as in Valori et al.
    (2010). The default parameters are those of Wiegelmann et al. (2006) in
    units of d (R = 2.2, a = 0.7, d = 1, L = 2).

    ``n_turns`` is *nominal*: Eqs. (7)–(9) assume a uniform B_θ inside the
    tube, which only holds for weak twist. The actual twist of the core
    (T_w of the lines near the axis) is ≈ −0.46, −0.69 and −0.78 turns with
    n_turns = 0.5, 1 and 1.5; with n_turns = 1, I0/I = −3.46 and T_w ≈ −0.69 turns,
    consistent with Wiegelmann et al. (2006): I0/I = −3.83 and −1.41π (−0.7 turns).

    The model is force-free only to first order in a/R and a/L (the field
    is an approximate equilibrium): ``meta["approx"]`` says so. With 64² points,
    CW_sin = 0.063 (0.056 with 96²; Low & Lou gives 0.017), and it is minimal with the
    equilibrium current of Eq. (6): the residual is the first-order error
    of the model, not of the implementation.

    Grid: x, y ∈ [−``half_width``, ``half_width``] (units of d), z ≥ 0 with the
    same step; the box is ``box_mm`` Mm wide and the maximum |B_z| at the base is
    ``bmax`` G. ``current_factor`` multiplies the equilibrium current I
    (only for tests: with values ≠ 1 the torus is not in equilibrium).
    """
    h = 2 * half_width / (nx - 1)
    xs = -half_width + h * np.arange(nx)
    ys = -(ny - 1) * h / 2 + h * np.arange(ny)
    zs = h * np.arange(nz)
    X, Y, Z = np.meshgrid(xs, ys, zs, indexing="ij")
    q = 1.0
    li = 0.5
    I = 8 * np.pi * q * L * R * (R * R + L * L) ** -1.5 / (np.log(8 * R / a) - 1.5 + li / 2)  # Eq. (6)
    I *= current_factor
    Nt = np.pi * n_turns / np.arccos(d / R)  # Eq. (7)
    I0 = handedness * I * R * R / (a * a * Nt)  # Eqs. (8)-(9): N_t = I R / (2π a² |B_θ|), B_θ = I0/(2πR)

    zd = Z + d
    rp = np.sqrt(Y ** 2 + zd ** 2)  # Eq. (19)
    rho = np.sqrt(X ** 2 + (rp - R) ** 2)  # Eq. (18)
    th = (np.zeros_like(X), -zd / rp, Y / rp)  # Eq. (17)

    # B_q, Eqs. (20)-(21)
    Bq = [np.zeros_like(X) for _ in range(3)]
    for s in (+1, -1):
        r = (X - s * L, Y, zd)
        n3 = (r[0] ** 2 + r[1] ** 2 + r[2] ** 2) ** 1.5
        for i in range(3):
            Bq[i] += s * q * r[i] / n3

    # B_θ, Eq. (16)
    chi = (rho < a).astype(float)
    inner = 1 / R ** 2 + 2 * chi * (I * I / (a * a * I0 * I0)) * (1 - rho ** 2 / a ** 2)
    bth = I0 / (2 * np.pi) * (np.sqrt(inner) + 1 / rp - 1 / R)

    # B_I, Eq. (22), with centred derivatives of A_I (Eqs. 25-31)
    cI = I / (2 * np.pi)
    eps = 1e-5
    A = _td_vector_potential(X, rp, R, a, cI)
    dAdx = (_td_vector_potential(X + eps, rp, R, a, cI) - _td_vector_potential(X - eps, rp, R, a, cI)) / (2 * eps)
    dAdr = (_td_vector_potential(X, rp + eps, R, a, cI) - _td_vector_potential(X, rp - eps, R, a, cI)) / (2 * eps)
    bI_x = dAdr + A / rp
    bI_r = -dAdx  # along r̂⊥ = (0, y, z + d)/r⊥

    bx = Bq[0] + bI_x
    by = Bq[1] + bth * th[1] + bI_r * Y / rp
    bz = Bq[2] + bth * th[2] + bI_r * zd / rp
    s = bmax / np.abs(bz[:, :, 0]).max()
    unit = box_mm / (2 * half_width)  # Mm per unit of d
    f = VectorField3D(bx * s, by * s, bz * s, dx=h * unit,
                      origin=(xs[0] * unit, ys[0] * unit, 0.0),
                      meta={"method": "Titov-Démoulin (1999)", "approx": "libre de fuerzas a primer orden en a/R",
                            "R": R, "a": a, "d": d, "L": L, "n_turns": n_turns, "handedness": handedness,
                            "I": I, "I0": I0, "unit_mm": unit})
    return f

    def field_magnetogram(field: VectorField3D, noise: float = 0.0, noise_t: float | None = None,
                      rng: int | None = 0) -> Magnetogram:
    """z = 0 magnetogram of a 3D field with Gaussian noise.

    ``noise``: σ (G) of B_z; ``noise_t``: σ of B_x and B_y (default = ``noise``).
    With ``noise_t`` > ``noise`` HMI is mimicked, whose transverse field is much
    noisier than the longitudinal one. The error maps are stored in the
    magnetogram (``*_err``).
    """
    noise_t = noise if noise_t is None else noise_t
    bx, by, bz = (c.copy() for c in field.bottom())
    if noise > 0 or noise_t > 0:
        g = np.random.default_rng(rng)
        bx += g.normal(0, noise_t, bx.shape)
        by += g.normal(0, noise_t, by.shape)
        bz += g.normal(0, noise, bz.shape)
    full = lambda s: np.full(bz.shape, float(s))
    return Magnetogram(bz=bz, bx=bx, by=by, dx=field.dx, origin=field.origin[:2],
                       bx_err=full(noise_t), by_err=full(noise_t), bz_err=full(noise),
                       meta={"source": field.meta.get("method", "analítico"), "noise": noise,
                             "noise_t": noise_t, **field.meta})

    - ``nz``: levels in height.

    def spread_indices(points: np.ndarray, k: int) -> np.ndarray:
    """k well-spread indices of ``points`` (farthest-point sampling)."""
    points = np.asarray(points)[:, :2]
    k = min(k, len(points))
    if k == 0:
        return np.array([], dtype=int)
    c = points.mean(axis=0)
    idx = [int(np.argmin(np.linalg.norm(points - c, axis=1)))]
    d = np.linalg.norm(points - points[idx[0]], axis=1)
    for _ in range(k - 1):
        i = int(np.argmax(d))
        idx.append(i)
        d = np.minimum(d, np.linalg.norm(points - points[i], axis=1))
    # Stable order (by x) so that the numbering is readable
    idx = sorted(idx, key=lambda i: (points[i, 0], points[i, 1]))
    return np.array(idx)

            means[i], stds[i] = v.mean(), v.std()
    return means, stds

    solarff.aia ¶

    Comparison of the field lines with the coronal loops observed by SDO/AIA.

    External validation with real data (as in Warren et al. 2018): the model lines are projected onto an AIA EUV image and the misalignment angle between the tangent of the projected line and the local orientation of the bright structures of the image (loops) is measured point by point.

    Steps:

    1. download_aia — level-1 image closest to the HMI observation (aia.lev1_euv_12s, requires drms and an email registered at JSOC, which is only passed as an argument and never stored).
    2. project_points — from the local Cartesian coordinates of the model (x, y, z in Mm on the CEA projection of the SHARP) to AIA pixels: CEA → Carrington (longitude, latitude) with the WCS of the SHARP, radius R☉ + z, and transformation to the SDO viewpoint with sunpy.coordinates. The Cartesian box is a local (flat) approximation of the sphere, the usual one in NLFFF with SHARP CEA.
    3. ridge_orientation — local orientation of the bright ridges (loops) from the Hessian matrix of the smoothed image: the direction of the loop is the eigenvector of the eigenvalue of smallest modulus, and the ridge strength is −λ_min (negative curvature = bright ridge).
    4. line_misalignment — for each line, median of the angle between its projected tangent and the ridge orientation at the points where there is a clear ridge (above a strength percentile).

    The angle lies in [0°, 90°] (orientations have no sense); a line unrelated to the image gives 45° on average.

    def download_aia(time_utc: str, wavelengths=(171, 193), email: str | None = None,
                 out_dir: str = "data/aia", window: str = "2m") -> dict:

    import pandas as pd
    os.makedirs(out_dir, exist_ok=True)
    t = pd.Timestamp(time_utc.replace("Z", ""))
    half = pd.Timedelta(window) if window[-1] != "m" else pd.Timedelta(minutes=int(window[:-1]))
    q0 = (t - half).strftime("%Y-%m-%dT%H:%M:%SZ")
    span = f"{int(2 * half.total_seconds())}s"
    c = drms.Client(email=email) if email else drms.Client()
    out = {}
    for w in wavelengths:
        k = c.query(f"{AIA_SERIES}[{q0}/{span}][{w}]", key="T_REC,T_OBS,QUALITY")
        k = k[k.QUALITY == 0].copy()
        if k.empty:
            raise RuntimeError(f"No hay imágenes AIA {w} Å con QUALITY = 0 cerca de {time_utc}")
        k["dt"] = (pd.to_datetime(k.T_OBS.str.replace("Z", "")) - t).abs()
        rec = k.sort_values("dt").iloc[0]
        tag = rec.T_REC.replace(":", "").replace("-", "")
        existing = glob.glob(os.path.join(out_dir, f"*{w}*{tag[:15]}*.fits"))
        if existing:
            out[w] = existing[0]
            continue
        if not email:
            raise ValueError("Hace falta un email registrado en JSOC para descargar (argumento email)")
        r = c.export(f"{AIA_SERIES}[{rec.T_REC}][{w}]{{image}}", method="url", protocol="fits")
        r.wait()
        d = r.download(out_dir)
        path = d.download[0]
        new = os.path.join(out_dir, f"aia.lev1_euv_12s.{w}.{tag}.fits")
        os.replace(path, new)
        out[w] = new
    return out

    def ridge_orientation(img: np.ndarray, sigma: float = 2.0, log: bool = True):
    """Orientation (rad, in [0, π)) and strength of the bright ridges of an image.

    Hessian of the image (of its logarithm, by default) smoothed with a
    Gaussian of width ``sigma`` pixels. On a bright ridge the transverse
    curvature is strongly negative (λ_min ≪ 0) and the longitudinal one nearly
    zero: the ridge direction is the eigenvector of λ_max. Strength =
    max(−λ_min, 0). ``img[row=y, column=x]``; the angle is measured from +x.
    """
    from scipy.ndimage import gaussian_filter
    a = np.log(np.clip(img, np.nanpercentile(img, 1) + 1e-3, None)) if log else img.astype(float)
    a = np.nan_to_num(a)
    hxx = gaussian_filter(a, sigma, order=(0, 2))
    hyy = gaussian_filter(a, sigma, order=(2, 0))
    hxy = gaussian_filter(a, sigma, order=(1, 1))
    tr, det = hxx + hyy, hxx * hyy - hxy * hxy
    disc = np.sqrt(np.maximum(tr * tr / 4 - det, 0))
    lmin = tr / 2 - disc
    lmax = tr / 2 + disc
    # eigenvector of lmax: (hxy, lmax − hxx) or (lmax − hyy, hxy); the one with the larger norm is taken
    # (one of the two vanishes when the Hessian is diagonal)
    ax, ay = hxy, lmax - hxx
    bx, by = lmax - hyy, hxy
    use_a = ax * ax + ay * ay >= bx * bx + by * by
    theta = np.mod(np.arctan2(np.where(use_a, ay, by), np.where(use_a, ax, bx)), np.pi)
    strength = np.maximum(-lmin, 0.0) * sigma ** 2  # scale-normalised
    return theta, strength

    def line_misalignment(pix_lines: list, theta: np.ndarray, strength: np.ndarray, threshold: float,
                      min_points: int = 5, step_px: float = 2.0) -> np.ndarray:
    """Median misalignment angle (degrees) of each projected line.

    ``pix_lines``: list of (N, 2) arrays of AIA pixels. Each line is
    resampled every ``step_px`` pixels along its projection; the points with
    ridge strength ≥ ``threshold`` are used. Lines with fewer than
    ``min_points`` valid points give NaN.
    """
    out = np.full(len(pix_lines), np.nan)
    for k, P in enumerate(pix_lines):
        if len(P) < 3:
            continue
        seg = np.hypot(*np.diff(P, axis=0).T)
        s = np.concatenate([[0], np.cumsum(seg)])
        if s[-1] < 3 * step_px:
            continue
        si = np.arange(0, s[-1], step_px)
        x, y = np.interp(si, s, P[:, 0]), np.interp(si, s, P[:, 1])
        tx, ty = np.gradient(x), np.gradient(y)
        ang = np.mod(np.arctan2(ty, tx), np.pi)
        th, st = _sample(theta, x, y), _sample(strength, x, y)
        ok = np.isfinite(th) & (st >= threshold) & (np.hypot(tx, ty) > 1e-6)
        if ok.sum() < min_points:
            continue
        d = np.abs(ang[ok] - th[ok])
        d = np.minimum(d, np.pi - d)
        out[k] = np.degrees(np.median(d))
    return out

    def aia_validation(lines, conf, reliable, files: dict, sharp_fits: str, other_lines: dict | None = None,
                   sigma: float = 2.0, pct: float = 90, n_boot: int = 2000, seed: int = 0) -> dict:
    """Compare the lines of a model with the AIA loops (see the module docstring).

    ``lines``: model lines (objects with ``points`` and ``apex``); ``conf`` and
    ``reliable``: their confidence; ``files``: {wavelength: AIA FITS};
    ``other_lines``: {name: lines traced from the same seeds in other
    models (e.g. potential)}. Returns ``{"summary": ..., "plot": ...}``:
    the summary (JSON) and what the figure needs.

    Since in a region the confidence is usually strongly tied to the height of the
    line, the comparison is repeated at equal height (terciles of the maximum
    height) and with a regression angle ~ reliable + log(1 + height).
    """
    from scipy.stats import mannwhitneyu, spearmanr
    rng = np.random.default_rng(seed)
    other_lines = other_lines or {}
    conf = np.asarray(conf, float)
    rel = np.asarray(reliable, bool)
    apex = np.array([l.apex for l in lines])
    S = {"n_lines": len(lines), "reliable_frac": float(rel.mean()), "sigma_px": sigma, "ridge_pct": pct,
         "by_wavelength": {}}
    plot = {}
    for w, path in files.items():
        m = load_aia(path)
        pix = [project_points(np.asarray(l.points), sharp_fits, m) for l in lines]
        allp = np.vstack(pix)
        x0, y0 = np.maximum(np.floor(allp.min(0) - 30).astype(int), 0)
        x1, y1 = np.ceil(allp.max(0) + 30).astype(int)
        sub = m.data[y0:y1, x0:x1]
        shift = np.array([x0, y0])
        pl = [p - shift for p in pix]
        th, st = ridge_orientation(sub, sigma=sigma)
        thr = float(np.percentile(st, pct))
        ang = line_misalignment(pl, th, st, thr)
        ang_o = {k: line_misalignment([project_points(np.asarray(l.points), sharp_fits, m) - shift for l in ls],
                                      th, st, thr) for k, ls in other_lines.items()}
        ok = np.isfinite(ang)
        r = {"file": os.path.basename(path), "date": m.date.isot, "n_measured": int(ok.sum()),
             "median_reliable": float(np.nanmedian(ang[rel])), "median_unreliable": float(np.nanmedian(ang[~rel])),
             "n_reliable": int((ok & rel).sum()), "n_unreliable": int((ok & ~rel).sum())}
        r["diff"], r["diff_ci"] = _boot_median_diff(ang[rel], ang[~rel], n_boot, rng)
        r["p_mannwhitney"] = float(mannwhitneyu(ang[ok & rel], ang[ok & ~rel], alternative="less").pvalue)
        r["spearman_conf"] = float(spearmanr(conf[ok], ang[ok])[0])
        # Height control
        la = np.log1p(apex[ok])
        X = np.column_stack([np.ones(ok.sum()), rel[ok].astype(float), la])
        y = ang[ok]
        beta = np.linalg.lstsq(X, y, rcond=None)[0]
        bs = [np.linalg.lstsq(X[i], y[i], rcond=None)[0][1]
              for i in (rng.integers(0, y.size, y.size) for _ in range(n_boot))]
        r["ols_reliable"] = float(beta[1])
        r["ols_reliable_ci"] = [float(np.percentile(bs, 2.5)), float(np.percentile(bs, 97.5))]
        Z = np.column_stack([np.ones_like(la), la])
        res_c = conf[ok] - Z @ np.linalg.lstsq(Z, conf[ok], rcond=None)[0]
        res_a = y - Z @ np.linalg.lstsq(Z, y, rcond=None)[0]
        r["spearman_partial"] = float(spearmanr(res_c, res_a)[0])
        r["spearman_partial_p"] = float(spearmanr(res_c, res_a)[1])
        q = np.percentile(apex[ok], [100 / 3, 200 / 3])
        terc = np.digitize(apex, q)
        r["terciles"] = []
        for i in range(3):
            s = ok & (terc == i)
            row = {"apex_mm": [float(apex[s].min()), float(apex[s].max())],
                   "n_reliable": int((s & rel).sum()), "n_unreliable": int((s & ~rel).sum()),
                   "median_reliable": float(np.median(ang[s & rel])) if (s & rel).any() else None,
                   "median_unreliable": float(np.median(ang[s & ~rel])) if (s & ~rel).any() else None}
            if row["n_reliable"] >= 5 and row["n_unreliable"] >= 5:
                row["diff"], row["diff_ci"] = _boot_median_diff(ang[s & rel], ang[s & ~rel], n_boot, rng)
            for k, v in ang_o.items():
                row[f"nlfff_minus_{k}_reliable"] = _paired((ang - v)[s & rel])
                row[f"nlfff_minus_{k}_unreliable"] = _paired((ang - v)[s & ~rel])
            r["terciles"].append(row)
        r["models"] = {}
        for k, v in ang_o.items():
            b = ok & np.isfinite(v)
            r["models"][k] = {"all": _paired((ang - v)[b]), "reliable": _paired((ang - v)[b & rel]),
                              "unreliable": _paired((ang - v)[b & ~rel]),
                              "median_nlfff": float(np.median(ang[b])), "median_other": float(np.median(v[b]))}
        # Sensitivity to the loop detection
        sens = []
        for sg in (1.5, 2.0, 3.0):
            th2, st2 = (th, st) if sg == sigma else ridge_orientation(sub, sigma=sg)
            for pc in (85, 90, 95):
                a2 = line_misalignment(pl, th2, st2, float(np.percentile(st2, pc)))
                mid = np.isfinite(a2) & (terc == 1)
                sens.append({"sigma": sg, "pct": pc, "median_reliable": float(np.nanmedian(a2[rel])),
                             "median_unreliable": float(np.nanmedian(a2[~rel])),
                             "mid_reliable": float(np.nanmedian(a2[mid & rel])) if (mid & rel).any() else None,
                             "mid_unreliable": float(np.nanmedian(a2[mid & ~rel])) if (mid & ~rel).any() else None})
        r["sensitivity"] = sens
        S["by_wavelength"][str(w)] = r
        plot[w] = {"cutout": sub, "pix": pl, "angles": ang, "date": m.date.isot[:19]}
    return {"summary": S, "plot": {"wavelengths": plot, "reliable": rel}}
