    solarff.lfff ¶

    Linear force-free field (LFFF): \(\nabla\times\mathbf{B} = \alpha\mathbf{B}\) with constant \(\alpha\).

    Fourier solution (Nakagawa & Raadu 1972; Alissandrakis 1981) with periodic lateral conditions. For each wavenumber \(\mathbf{k} = (k_x, k_y)\):

    \[ \begin{aligned} l &= \sqrt{k^2 - \alpha^2} \\ \hat B_z(z) &= \hat B_z(0)\, e^{-l z} \\ \hat B_x(z) &= -i\,\frac{k_x l - \alpha k_y}{k^2}\, \hat B_z(z) \\ \hat B_y(z) &= -i\,\frac{k_y l + \alpha k_x}{k^2}\, \hat B_z(z) \end{aligned} \]

    Only requires \(B_z\) at the photosphere. Solutions that decay with height require \(|\alpha| < k_{\min} = 2\pi / L_{\max}\) (\(L_{\max}\): longer side of the periodic domain). \(\alpha = 0\) gives the potential field.

    def alpha_max(mag: Magnetogram, pad: float = 1.0) -> float:
    """Maximum admissible |α| (Mm^-1) for the domain (with the given padding factor)."""
    L = max(mag.shape) * pad * mag.dx
    return 2 * np.pi / L

    def lfff(mag: Magnetogram, alpha: float, nz: int, pad: float = 1.0,
         heights: np.ndarray | None = None) -> VectorField3D:
    """Linear force-free extrapolation.

    **Parameters**

    - ``mag``: magnetogram (only ``bz`` is used).
    - ``alpha``: force-free parameter in Mm^-1 (positive = positive helicity /
        right-handed chirality).
    - ``nz``: number of levels in height (step = ``mag.dx``).
    - ``pad``: zero-padding factor (>= 1). Reduces the interaction with the
        periodic copies, but also reduces the maximum admissible α.
    - ``heights``: explicit heights in Mm (if given, ``nz`` is ignored; the returned
        field no longer has a uniform step in z — useful only to evaluate
        specific layers).
    """
    if pad < 1:
        raise ValueError("pad debe ser >= 1")
    bz0, (ox, oy) = _padded(mag.bz, pad)
    nx, ny = mag.shape
    px, py = bz0.shape
    h = mag.dx

    amax = 2 * np.pi / (max(px, py) * h)
    if abs(alpha) >= amax:
        raise ValueError(
            f"|alpha| = {abs(alpha):.4g} Mm^-1 >= alpha_max = {amax:.4g} Mm^-1 para este "
            f"dominio: la solución no decae con la altura. Reduce alpha o pad.")

    kx = 2 * np.pi * np.fft.fftfreq(px, d=h)
    ky = 2 * np.pi * np.fft.fftfreq(py, d=h)
    KX, KY = np.meshgrid(kx, ky, indexing="ij")
    k2 = KX ** 2 + KY ** 2
    k2_safe = np.where(k2 == 0, 1.0, k2)
    l = np.sqrt(np.maximum(k2 - alpha ** 2, 0.0))

    bz_hat = np.fft.fft2(bz0)
    fx = -1j * (KX * l - alpha * KY) / k2_safe
    fy = -1j * (KY * l + alpha * KX) / k2_safe
    fx[0, 0] = fy[0, 0] = 0.0
    if alpha != 0:
        # A uniform field does not satisfy ∇×B = αB with α ≠ 0: the k=0 mode is removed.
        bz_hat[0, 0] = 0.0

    z = np.asarray(heights, dtype=float) if heights is not None else h * np.arange(nz)
    bx = np.empty((nx, ny, z.size))
    by = np.empty_like(bx)
    bz = np.empty_like(bx)
    for iz, zz in enumerate(z):
        decay = bz_hat * np.exp(-l * zz)
        bz[:, :, iz] = np.fft.ifft2(decay).real[ox:ox + nx, oy:oy + ny]
        bx[:, :, iz] = np.fft.ifft2(fx * decay).real[ox:ox + nx, oy:oy + ny]
        by[:, :, iz] = np.fft.ifft2(fy * decay).real[ox:ox + nx, oy:oy + ny]

    method = "potencial" if alpha == 0 else "LFFF"
    return VectorField3D(bx, by, bz, dx=h, origin=(*mag.origin, 0.0),
                         meta={"method": method, "alpha": float(alpha), "pad": pad})

    linear_mhs(mag: Magnetogram, alpha: float, nz: int, a: float, kappa: float, pad: float = 1.0) -> VectorField3D

    Linear magnetohydrostatic field of Low (1991): an exact equilibrium whose lowest layers are not force-free.

    \[ \nabla\times\mathbf{B} = \alpha\mathbf{B} + a\,e^{-\kappa z}\,\nabla B_z\times\hat{\mathbf{z}}, \qquad \nabla\cdot\mathbf{B} = 0 . \]

    The second term is a horizontal current perpendicular to the field, balanced by pressure and gravity; it decays with the scale height \(1/\kappa\), above which the field is linear force-free. For each horizontal wavenumber \(k\) the vertical component is \(\hat B_z(z) = \hat B_z(0)\,\varphi_k(z)\) with

    \[ \varphi_k(z) = \frac{J_s(q\,e^{-\kappa z/2})}{J_s(q)}, \qquad s = \frac{2\sqrt{k^2-\alpha^2}}{\kappa}, \qquad q = \frac{2k\sqrt{a}}{\kappa}, \]

    and the horizontal components follow from \(\nabla\cdot\mathbf{B} = 0\) and \(J_z = \alpha B_z\): \(\hat B_x = i\,(k_x\varphi' + \alpha k_y\varphi)\,\hat B_z(0)/k^2\) and \(\hat B_y = i\,(k_y\varphi' - \alpha k_x\varphi)\,\hat B_z(0)/k^2\). With \(a = 0\) it is the LFFF of lfff. The Bessel functions are evaluated through \({}_0F_1\), which stays finite for the high orders of the small scales.

    Parameters

    • mag: magnetogram (only bz is used).
    • alpha: force-free parameter of the upper layers, in Mm^-1.
    • nz: number of levels in height (step = mag.dx).
    • a: amplitude of the non-force-free current at the base, 0 <= a < 1.
    • kappa: inverse scale height of that current, in Mm^-1.
    • pad: zero-padding factor, as in lfff.

    def linear_mhs(mag: Magnetogram, alpha: float, nz: int, a: float, kappa: float, pad: float = 1.0) -> VectorField3D:
    r"""Linear magnetohydrostatic field of Low (1991): an exact equilibrium whose lowest layers are not force-free.

    $$
    \nabla\times\mathbf{B} = \alpha\mathbf{B} + a\,e^{-\kappa z}\,\nabla B_z\times\hat{\mathbf{z}},
    \qquad \nabla\cdot\mathbf{B} = 0 .
    $$

    The second term is a horizontal current perpendicular to the field, balanced by pressure and gravity; it
    decays with the scale height $1/\kappa$, above which the field is linear force-free. For each horizontal
    wavenumber $k$ the vertical component is $\hat B_z(z) = \hat B_z(0)\,\varphi_k(z)$ with

    $$
    \varphi_k(z) = \frac{J_s(q\,e^{-\kappa z/2})}{J_s(q)}, \qquad
    s = \frac{2\sqrt{k^2-\alpha^2}}{\kappa}, \qquad q = \frac{2k\sqrt{a}}{\kappa},
    $$

    and the horizontal components follow from $\nabla\cdot\mathbf{B} = 0$ and $J_z = \alpha B_z$:
    $\hat B_x = i\,(k_x\varphi' + \alpha k_y\varphi)\,\hat B_z(0)/k^2$ and
    $\hat B_y = i\,(k_y\varphi' - \alpha k_x\varphi)\,\hat B_z(0)/k^2$. With $a = 0$ it is the LFFF of ``lfff``.
    The Bessel functions are evaluated through ${}_0F_1$, which stays finite for the high orders of the small scales.

    **Parameters**

    - ``mag``: magnetogram (only ``bz`` is used).
    - ``alpha``: force-free parameter of the upper layers, in Mm^-1.
    - ``nz``: number of levels in height (step = ``mag.dx``).
    - ``a``: amplitude of the non-force-free current at the base, 0 <= a < 1.
    - ``kappa``: inverse scale height of that current, in Mm^-1.
    - ``pad``: zero-padding factor, as in ``lfff``.
    """
    from scipy.special import hyp0f1
    if not 0 <= a < 1:
        raise ValueError("a debe estar en [0, 1)")
    if kappa <= 0:
        raise ValueError("kappa debe ser positivo")
    bz0, (ox, oy) = _padded(mag.bz, pad)
    nx, ny = mag.shape
    px, py = bz0.shape
    h = mag.dx
    amax = 2 * np.pi / (max(px, py) * h)
    if abs(alpha) >= amax:
        raise ValueError(f"|alpha| = {abs(alpha):.4g} Mm^-1 >= alpha_max = {amax:.4g} Mm^-1 para este dominio")

    kx = 2 * np.pi * np.fft.fftfreq(px, d=h)
    ky = 2 * np.pi * np.fft.fftfreq(py, d=h)
    KX, KY = np.meshgrid(kx, ky, indexing="ij")
    k2 = KX ** 2 + KY ** 2
    k2_safe = np.where(k2 == 0, 1.0, k2)
    l = np.sqrt(np.maximum(k2 - alpha ** 2, 0.0))
    s = 2 * l / kappa
    q2 = 4 * a * k2 / kappa ** 2
    g1 = hyp0f1(s + 1, -q2 / 4)
    if np.any(g1 <= 0):
        raise ValueError("la solución no es regular para estos a, kappa y alpha (J_s(q) <= 0 en algún modo)")

    bz_hat = np.fft.fft2(bz0)
    if alpha != 0 or a != 0:
        bz_hat[0, 0] = 0.0   # a uniform field is not a solution
    bx = np.empty((nx, ny, nz))
    by = np.empty_like(bx)
    bz = np.empty_like(bx)
    for iz in range(nz):
        zz = h * iz
        u2 = np.exp(-kappa * zz)
        decay = np.exp(-l * zz)
        phi = decay * hyp0f1(s + 1, -q2 * u2 / 4) / g1
        dphi = -l * phi + decay * (q2 * u2 * kappa / 4) * hyp0f1(s + 2, -q2 * u2 / 4) / ((s + 1) * g1)
        fx = 1j * (KX * dphi + alpha * KY * phi) / k2_safe
        fy = 1j * (KY * dphi - alpha * KX * phi) / k2_safe
        fx[0, 0] = fy[0, 0] = 0.0
        bz[:, :, iz] = np.fft.ifft2(bz_hat * phi).real[ox:ox + nx, oy:oy + ny]
        bx[:, :, iz] = np.fft.ifft2(bz_hat * fx).real[ox:ox + nx, oy:oy + ny]
        by[:, :, iz] = np.fft.ifft2(bz_hat * fy).real[ox:ox + nx, oy:oy + ny]
    return VectorField3D(bx, by, bz, dx=h, origin=(*mag.origin, 0.0),
                         meta={"method": "MHS lineal", "alpha": float(alpha), "a": float(a), "kappa": float(kappa), "pad": pad})

    def alpha_from_jz(mag: Magnetogram, bz_min: float = 100.0, bt_min: float = 100.0) -> float:
    r"""$\alpha_{\rm av} = \sum J_zB_z / \sum B_z^2$ (Hagino & Sakurai 2004), with $J_z = (\nabla\times\mathbf{B})_z$.

    Only uses pixels with |Bz| > ``bz_min`` and |Bt| > ``bt_min`` (G) to
    avoid the noise of the weak fields.
    """
    mag._require_vector()
    m = _mask(mag, bz_min, bt_min)
    jz = mag.jz()
    return float((jz[m] * mag.bz[m]).sum() / (mag.bz[m] ** 2).sum())

    def alpha_best_fit(mag: Magnetogram, pad: float = 1.0, bt_min: float = 100.0,
                   n_scan: int = 41, return_curve: bool = False):
    r"""α that best reproduces the observed transverse field with an LFFF.

    For each α, (Bx, By) of the LFFF at z = 0 is computed and
    $\sum |\mathbf{B}_{t,\rm obs} - \mathbf{B}_{t,\rm LFFF}|^2$ is minimised over the pixels with |Bt| > ``bt_min``.
    Requires a vector magnetogram with the 180° ambiguity resolved (SHARP
    already is).
    """
    mag._require_vector()
    amax = 0.98 * alpha_max(mag, pad)
    m = _mask(mag, 0.0, bt_min)
    if m.sum() < 10:
        m = np.ones_like(mag.bz, dtype=bool)

    def cost(a):
        f = lfff(mag, a, nz=1, pad=pad)
        dx = f.bx[:, :, 0] - mag.bx
        dy = f.by[:, :, 0] - mag.by
        return float(((dx ** 2 + dy ** 2)[m]).sum() / ((mag.bx ** 2 + mag.by ** 2)[m]).sum())

    grid = np.linspace(-amax, amax, n_scan)
    costs = np.array([cost(a) for a in grid])
    i = int(np.argmin(costs))
    lo, hi = grid[max(i - 1, 0)], grid[min(i + 1, n_scan - 1)]
    res = minimize_scalar(cost, bounds=(lo, hi), method="bounded", options={"xatol": 1e-6})
    best = float(res.x)
    if return_curve:
        return best, grid, costs
    return best

    def estimate_alpha(mag: Magnetogram, method: str = "fit", **kw) -> float:
    """Estimate α (Mm^-1). ``method``: ``"fit"`` (fit of Bt) or ``"jz"`` (α_av).

    The value is clipped to ±0.98·α_max so that the LFFF is computable.
    """
    if method == "fit":
        return alpha_best_fit(mag, **kw)
    if method == "jz":
        a = alpha_from_jz(mag, **kw)
        amax = 0.98 * alpha_max(mag)
        return float(np.clip(a, -amax, amax))
    raise ValueError("method debe ser 'fit' o 'jz'")

    solarff.preprocess ¶

    Preprocessing of the vector magnetogram (Wiegelmann, Inhester & Sakurai 2006).

    The photosphere is not force-free. Before the NLFFF the magnetogram is modified slightly so that it meets the Aly criteria (zero net force and torque), resembles the data and is smooth (closer to the chromospheric field). We minimise

    \[ L = \mu_1 L_1 + \mu_2 L_2 + \mu_3 L_3 + \mu_4 L_4 \]
    \[ \begin{aligned} L_1 &= \Big(\sum B_xB_z\Big)^2 + \Big(\sum B_yB_z\Big)^2 + \Big(\sum D\Big)^2 && \text{(net force)} \\ L_2 &= \Big(\sum x\,D\Big)^2 + \Big(\sum y\,D\Big)^2 + \Big(\sum (y\,B_xB_z - x\,B_yB_z)\Big)^2 && \text{(net torque)} \\ L_3 &= \sum |\mathbf{B} - \mathbf{B}_{\rm obs}|^2 && \text{(data)} \\ L_4 &= \sum |\Delta\mathbf{B}|^2 && \text{(smoothness)} \end{aligned} \]

    with \(D = B_z^2 - B_x^2 - B_y^2\) and the sums over the pixels of the magnetogram.

    Here each term is normalised (sums → means, divided by \(\langle B_{\rm obs}^2\rangle\)) so that the weights \(\mu_i\) are independent of the size and intensity of the magnetogram. Minimised with L-BFGS-B and analytical gradient.

    def preprocess(mag: Magnetogram, mu1: float = 1.0, mu2: float = 1.0,
               mu3: float = 1e-2, mu4: float = 1e-2, max_iter: int = 500,
               fix_bz: bool = False, verbose: bool = True) -> Magnetogram:
    """Return a preprocessed magnetogram (does not modify the original).

    ``mu3`` controls how far from the data it may move and ``mu4`` how much it is
    smoothed. ``fix_bz=True`` only modifies the transverse components (noisier
    in HMI).
    """
    mag._require_vector()
    nx, ny = mag.shape
    n = nx * ny
    s = float(np.abs(np.stack([mag.bx, mag.by, mag.bz])).max())
    obs = np.stack([mag.bx, mag.by, mag.bz]) / s
    b2m = float((obs ** 2).sum(axis=0).mean())
    L = max(nx, ny) / 2
    x, y = np.meshgrid((np.arange(nx) - (nx - 1) / 2) / L,
                       (np.arange(ny) - (ny - 1) / 2) / L, indexing="ij")

    def fun(v):
        b = v.reshape(3, nx, ny)
        if fix_bz:
            b = np.concatenate([b[:2], obs[2:3]])
        bx, by, bz = b
        d = bz ** 2 - bx ** 2 - by ** 2
        c = 1.0 / (n * b2m)
        # L1
        sxz, syz, sd = c * (bx * bz).sum(), c * (by * bz).sum(), c * d.sum()
        L1 = sxz ** 2 + syz ** 2 + sd ** 2
        g1x = 2 * c * (sxz * bz - 2 * sd * bx)
        g1y = 2 * c * (syz * bz - 2 * sd * by)
        g1z = 2 * c * (sxz * bx + syz * by + 2 * sd * bz)
        # L2
        tx, ty = c * (x * d).sum(), c * (y * d).sum()
        tz = c * (y * bx * bz - x * by * bz).sum()
        L2 = tx ** 2 + ty ** 2 + tz ** 2
        g2x = 2 * c * (-2 * (tx * x + ty * y) * bx + tz * y * bz)
        g2y = 2 * c * (-2 * (tx * x + ty * y) * by - tz * x * bz)
        g2z = 2 * c * (2 * (tx * x + ty * y) * bz + tz * (y * bx - x * by))
        # L3
        diff = b - obs
        L3 = (diff ** 2).sum() / (n * b2m)
        g3 = 2 * diff / (n * b2m)
        # L4
        lap = np.stack([_laplacian(c_) for c_ in b])
        L4 = (lap ** 2).sum() / (n * b2m)
        g4 = 2 * np.stack([_laplacian_adjoint(l_, (nx, ny)) for l_ in lap]) / (n * b2m)

        total = mu1 * L1 + mu2 * L2 + mu3 * L3 + mu4 * L4
        grad = (mu1 * np.stack([g1x, g1y, g1z]) + mu2 * np.stack([g2x, g2y, g2z])
                + mu3 * g3 + mu4 * g4)
        if fix_bz:
            grad[2] = 0.0
        return total, grad.ravel()

    res = minimize(fun, obs.ravel(), jac=True, method="L-BFGS-B",
                   options={"maxiter": max_iter})
    b = res.x.reshape(3, nx, ny) * s
    if fix_bz:
        b[2] = mag.bz
    out = Magnetogram(bz=b[2], bx=b[0], by=b[1], dx=mag.dx,
                      origin=mag.origin, meta={**mag.meta, "preprocessed": True})
    if verbose:
        a0, a1 = aly_criteria(mag), aly_criteria(out)
        print(f"  Preprocesado ({res.nit} it): eps_force {a0['eps_force']:.3f} -> "
              f"{a1['eps_force']:.3f}, eps_torque {a0['eps_torque']:.3f} -> {a1['eps_torque']:.3f}")
    return out

    solarff.nlfff_torch ¶

    NLFFF with exact gradient (automatic differentiation, PyTorch) and L-BFGS.

    Minimises the discrete functional

    \[ L = \Big\langle w\Big[\frac{|\mathbf{J}\times\mathbf{B}|^2}{B^2+\varepsilon^2} + (\nabla\cdot\mathbf{B})^2\Big]\Big\rangle_V + \nu\,\Big\langle W\,|\mathbf{B}_t - \mathbf{B}_{t,\rm obs}|^2\Big\rangle_{\rm base} \]

    with the same finite differences as the numpy version (2nd order, also at the edges). PyTorch computes the exact gradient of L with respect to ALL the unknowns, including the transverse field of the base when it is relaxed. This allows:

    • Relaxed base (Wiegelmann et al. 2010, 2012): B_t at z = 0 is no longer fixed and may adjust within the measurement noise, with more freedom where the transverse field is weak (\(W = |\mathbf{B}_{t,\rm obs}| / \max|\mathbf{B}_{t,\rm obs}|\)). B_z stays fixed.
    • L-BFGS with Wolfe line search: converges in far fewer iterations than gradient descent with a fixed step.

    The side and top faces are fixed to the initial (potential) field and the weighting function w is the same as in solarff.nlfff.

    def nlfff_lbfgs(mag: Magnetogram, nz: int, initial: VectorField3D | None = None,
                nd: int | None = None, relax_bottom: float | None = 0.1, eps: float = 0.03,
                smooth: float = 1e-2, div_weight: float = 1.0,
                max_iter: int = 2000, tol: float = 1e-6, chunk: int = 50,
                threads: int | None = None, verbose: bool = True,
                return_result: bool = False):
    """NLFFF extrapolation with exact gradient and L-BFGS.

    **Parameters**

    - ``mag``: vector magnetogram (ideally preprocessed).
    - ``nz``: levels in height.
    - ``initial``: initial field; its side and top faces stay fixed
        (by default the potential).
    - ``nd``: width of the buffer layer (by default ~10 % of the shorter side).
    - ``relax_bottom``: weight ν of the data term. With ``None`` the base stays fixed
        (as in the classic method). Higher values = base closer to the
        data; lower = more freedom to make it consistent.
    - ``eps``: regularisation near nulls (fraction of max|Bz|).
    - ``smooth``: weight λ of the smoothness penalty λ <|∇²B|²>. Centred
        differences do not "see" grid-scale oscillations
        (checkerboard); an optimiser with exact gradient exploits them
        to lower L without improving the physics. This term suppresses them.
    - ``div_weight``: weight w_d of the divergence term relative to the force term
        (w_f = 1), as in Wiegelmann et al. (2012). More weight = more
        solenoidal field (helicity less gauge dependent) at the cost of somewhat more
        Lorentz force. Default 1 (standard form). With 4 the
        thresholds of Thalmann et al. (2019) are met in HARP 377, but the
        lines reproduce the exact field worse and neither the energy nor the helicity
        get closer to the truth (``examples/09`` and ``11``), so it is used
        only as a sensitivity test.
    - ``max_iter``: maximum number of L-BFGS iterations.
    - ``tol``: stop if L decreases relatively by less than ``tol`` per iteration
        (average over a block of ``chunk`` iterations).
    - ``threads``: CPU threads for PyTorch (None = whatever PyTorch decides).
    """
    torch = _require_torch()
    if threads:
        torch.set_num_threads(threads)
    mag._require_vector()
    nx, ny = mag.shape
    h = mag.dx
    if initial is None:
        initial = potential_field(mag, nz)
    if initial.shape != (nx, ny, nz):
        raise ValueError(f"campo inicial con forma {initial.shape}, se esperaba {(nx, ny, nz)}")
    if nd is None:
        nd = max(4, min(nx, ny, nz) // 10)
    relax = relax_bottom is not None and relax_bottom > 0
    nu = float(relax_bottom) if relax else 0.0

    scale = float(np.abs(mag.bz).max()) or 1.0
    B0 = np.stack([initial.bx, initial.by, initial.bz]) / scale
    B0[0, :, :, 0] = mag.bx / scale
    B0[1, :, :, 0] = mag.by / scale
    B0[2, :, :, 0] = mag.bz / scale
    dt = torch.float64
    fixed = torch.tensor(B0, dtype=dt)
    w = torch.tensor(weight_function((nx, ny, nz), nd), dtype=dt)
    obs = torch.tensor(B0[:2, 1:-1, 1:-1, 0].copy(), dtype=dt)
    bt = np.hypot(B0[0, 1:-1, 1:-1, 0], B0[1, 1:-1, 1:-1, 0])
    W = torch.tensor(bt / bt.max() if bt.max() > 0 else np.ones_like(bt), dtype=dt)

    P = torch.tensor(B0[:, 1:-1, 1:-1, 1:-1].copy(), dtype=dt, requires_grad=True)
    params = [P]
    if relax:
        Pb = torch.tensor(B0[:2, 1:-1, 1:-1, 0].copy(), dtype=dt, requires_grad=True)
        params.append(Pb)
    z0 = 0 if relax else 1
    region = (slice(1, -1), slice(1, -1), slice(z0, -1))
    eps2 = float(eps) ** 2

    def assemble():
        B = fixed.clone()
        B[:, 1:-1, 1:-1, 1:-1] = P
        if relax:
            B[:2, 1:-1, 1:-1, 0] = Pb
        return B

    def loss_terms():
        B = assemble()
        dens = _density(torch, B, eps2, div_weight)
        lff = (w[region] * dens[region]).mean()
        ldata = nu * (W * ((Pb - obs) ** 2).sum(0)).mean() if relax else torch.zeros((), dtype=dt)
        if smooth:
            c = B[:, 1:-1, 1:-1, 1:-1]
            lap = (B[:, 2:, 1:-1, 1:-1] + B[:, :-2, 1:-1, 1:-1] + B[:, 1:-1, 2:, 1:-1]
                   + B[:, 1:-1, :-2, 1:-1] + B[:, 1:-1, 1:-1, 2:] + B[:, 1:-1, 1:-1, :-2] - 6 * c)
            lff = lff + smooth * (lap ** 2).sum(0).mean()
        return lff, ldata

    opt = torch.optim.LBFGS(params, lr=1.0, max_iter=chunk, history_size=20,
                            line_search_fn="strong_wolfe",
                            tolerance_grad=1e-14, tolerance_change=1e-16)
    n_eval = [0]

    def closure():
        opt.zero_grad()
        lff, ld = loss_terms()
        loss = lff + ld
        loss.backward()
        n_eval[0] += 1
        return loss

    with torch.no_grad():
        lff, ld = loss_terms()
    history = [(0, float(lff + ld), float(lff), float(ld))]
    t0 = time.time()
    it, converged, message = 0, False, "max_iter alcanzado"
    while it < max_iter:
        L_prev = history[-1][1]
        opt.step(closure)
        it = int(opt.state[opt._params[0]].get("n_iter", it + chunk))
        with torch.no_grad():
            lff, ld = loss_terms()
        L = float(lff + ld)
        history.append((it, L, float(lff), float(ld)))
        if verbose:
            print(f"  NLFFF(L-BFGS) it {it:5d}  L = {L:.4e}  (fuerza+div {float(lff):.3e}, datos {float(ld):.3e})"
                  f"  ({time.time() - t0:.1f}s)", flush=True)
        done_iters = max(it - history[-2][0], 1)
        rel = (L_prev - L) / (abs(L_prev) * done_iters) if L_prev else 0.0
        if rel < tol:
            converged, message = True, f"convergido: dL/L por iteracion = {rel:.2e}"
            break
    with torch.no_grad():
        B = assemble().numpy() * scale
    if verbose:
        print(f"  NLFFF(L-BFGS) terminado: {message}; {it} iteraciones, {n_eval[0]} evaluaciones, "
              f"L: {history[0][1]:.3e} -> {history[-1][1]:.3e} ({time.time() - t0:.1f}s)")
    out = VectorField3D(B[0], B[1], B[2], dx=h, origin=(*mag.origin, 0.0),
                        meta={"method": "NLFFF", "backend": "lbfgs", "iterations": it,
                              "evaluations": n_eval[0], "L_final": history[-1][1],
                              "L_initial": history[0][1], "nd": nd, "eps": eps,
                              "relax_bottom": relax_bottom if relax else None, "smooth": smooth,
                              "div_weight": div_weight,
                              "message": message})
    if return_result:
        # History compatible with the classic method: (iteration, L, step) with step = NaN
        hist = [(i, L, np.nan) for i, L, _, _ in history]
        res = NLFFFResult(out, hist, converged, message)
        res.terms = history
        return res
    return out

    solarff.nlfff ¶

    Nonlinear force-free field (NLFFF): \((\nabla\times\mathbf{B})\times\mathbf{B} = 0\), \(\nabla\cdot\mathbf{B} = 0\), \(\alpha = \alpha(\mathbf{r})\).

    Optimisation method (Wheatland, Sturrock & Roumeliotis 2000) with a weighting function at the boundaries (Wiegelmann 2004). We minimise

    \[ L = \int_V w\left[\frac{|(\nabla\times\mathbf{B})\times\mathbf{B}|^2}{B^2} + |\nabla\cdot\mathbf{B}|^2\right] dV \]

    iterating \(\mathbf{B} \leftarrow \mathbf{B} + \mu\,\mathbf{F}\), with

    \[ \begin{aligned} \mathbf{\Omega} &= \frac{(\nabla\times\mathbf{B})\times\mathbf{B} - (\nabla\cdot\mathbf{B})\,\mathbf{B}}{B^2} \\ \mathbf{F} &= w\left[\nabla\times(\mathbf{\Omega}\times\mathbf{B}) - \mathbf{\Omega}\times(\nabla\times\mathbf{B}) - \nabla(\mathbf{\Omega}\cdot\mathbf{B}) + \mathbf{\Omega}\,(\nabla\cdot\mathbf{B}) + \Omega^2\,\mathbf{B}\right] \\ &\quad + (\mathbf{\Omega}\times\mathbf{B})\times\nabla w + (\mathbf{\Omega}\cdot\mathbf{B})\,\nabla w \end{aligned} \]

    Boundary conditions: the vector magnetogram at z = 0; sides and top taken from the initial (potential) field. The function \(w = 1\) in the interior and drops to 0 (cosine profile) in a layer of nd points next to the side and top faces, so that the unobserved conditions have little influence.

    def weight_function(shape, nd: int) -> np.ndarray:
    """w(x,y,z): 1 in the interior, decays as a cosine to 0 over ``nd`` points at the side
    and top edges (not at the base)."""
    nx, ny, nz = shape

    def profile(n, both=True):
        p = np.ones(n)
        if nd <= 0:
            return p
        i = np.arange(nd)
        ramp = 0.5 * (1 - np.cos(np.pi * i / nd))  # 0 -> ~1
        p[n - nd:] = ramp[::-1]
        if both:
            p[:nd] = ramp
        return p

    return (profile(nx)[:, None, None] * profile(ny)[None, :, None]
            * profile(nz, both=False)[None, None, :])

    def nlfff(mag: Magnetogram, nz: int, initial: VectorField3D | None = None,
          nd: int | None = None, max_iter: int = 5000, tol: float = 1e-5,
          patience: int = 50, mu0: float | None = None, eps: float = 0.03,
          verbose: bool = True, return_result: bool = False):
    """NLFFF extrapolation by optimisation.

    **Parameters**

    - ``mag``: *vector* magnetogram (ideally preprocessed with
        ``solarff.preprocess``).
    - ``nz``: levels in height.
    - ``initial``: initial field (by default the potential). Its values on the
        side and top faces are kept fixed.
    - ``nd``: width (in points) of the buffer layer. By default
        ~10 % of the shorter side, minimum 4.
    - ``max_iter``: maximum number of accepted iterations.
    - ``tol``: stopping criterion: relative change of L per iteration averaged
        over ``patience`` iterations smaller than ``tol``.
    - ``mu0``: initial step (in units of h²); adapted automatically.
    - ``eps``: regularisation near magnetic nulls, as a fraction of max|Bz|.
        In the functional B² + ε² is used instead of B²: without it, the points with
        |B| ≈ 0 (frequent in real data) make the force blow up and the step
        μ drops to ~1e-7 h², stalling the optimisation. 0 = original method.
    - ``return_result``: if True returns ``NLFFFResult`` with the history.
    """
    mag._require_vector()
    nx, ny = mag.shape
    h = mag.dx
    if initial is None:
        initial = potential_field(mag, nz)
    if not np.allclose(initial.origin[:2], mag.origin):
        raise ValueError("El campo inicial y el magnetograma tienen distinto origen")
    if initial.shape != (nx, ny, nz):
        raise ValueError(f"campo inicial con forma {initial.shape}, se esperaba {(nx, ny, nz)}")
    if nd is None:
        nd = max(4, min(nx, ny, nz) // 10)

    # Normalisation: the functional is scale invariant in Ω, but it is convenient
    # to work with B ~ O(1).
    scale = float(np.abs(mag.bz).max()) or 1.0
    b = [initial.bx / scale, initial.by / scale, initial.bz / scale]
    b = [c.copy() for c in b]
    b[0][:, :, 0] = mag.bx / scale
    b[1][:, :, 0] = mag.by / scale
    b[2][:, :, 0] = mag.bz / scale

    w = weight_function((nx, ny, nz), nd)
    gw = np.gradient(w, h, edge_order=2)

    eps2 = float(eps) ** 2  # B is normalised to max|Bz| = 1
    interior = (slice(1, -1), slice(1, -1), slice(1, -1))
    mu = (mu0 if mu0 is not None else 0.1) * h * h

    L, F = _functional_and_force(b, w, gw, h, eps2=eps2)
    history = [(0, L, mu)]
    it, rejected = 0, 0
    t0 = time.time()
    converged, message = False, "max_iter alcanzado"

    while it < max_iter:
        trial = [c.copy() for c in b]
        for i in range(3):
            trial[i][interior] += mu * F[i][interior]
        L_new, F_new = _functional_and_force(trial, w, gw, h, eps2=eps2)
        if np.isfinite(L_new) and L_new < L:
            b, F = trial, F_new
            it += 1
            history.append((it, L_new, mu))
            L = L_new
            mu *= 1.01
            rejected = 0
            if it > patience:
                L_old = history[-patience - 1][1]
                rel = (L_old - L) / (L_old * patience)
                if rel < tol:
                    converged, message = True, f"convergido: dL/L por iteracion = {rel:.2e}"
                    break
            if verbose and it % 200 == 0:
                print(f"  NLFFF it {it:5d}  L = {L:.4e}  mu/h^2 = {mu / h / h:.3e}  "
                      f"({time.time() - t0:.1f}s)")
        else:
            mu *= 0.5
            rejected += 1
            if rejected > 60 or mu < 1e-12 * h * h:
                converged, message = True, "paso mu demasiado pequeno (minimo local)"
                break

    if verbose:
        print(f"  NLFFF terminado: {message}; {it} iteraciones, L: "
              f"{history[0][1]:.3e} -> {L:.3e} ({time.time() - t0:.1f}s)")

    out = VectorField3D(b[0] * scale, b[1] * scale, b[2] * scale, dx=h, origin=(*mag.origin, 0.0),
                        meta={"method": "NLFFF", "iterations": it, "L_final": L,
                              "L_initial": history[0][1], "nd": nd, "eps": eps, "message": message})
    if return_result:
        return NLFFFResult(out, history, converged, message)
    return out

    solarff.grad_rubin ¶

    NLFFF by the Grad–Rubin method (Grad & Rubin 1958; Amari et al. 2006; Wheatland 2007).

    It belongs to a different family from the optimisation method: instead of minimising a functional, it imposes the equations by iterating current and field:

    1. \(\alpha\) at the base: \(\alpha_0 = (\nabla\times\mathbf{B})_z / B_z\) on ONE polarity (P: \(B_z > 0\), N: \(B_z < 0\)).
    2. \(\alpha(\mathbf{r})\): each point inherits \(\alpha_0\) from the footpoint of its field line on that polarity (\(\alpha = 0\) if the line does not reach it).
    3. \(\mathbf{J} = \alpha\,\mathbf{B}\), cleaned so that \(\nabla\cdot\mathbf{J} = 0\).
    4. \(\mathbf{B} = \mathbf{B}_p + \nabla\times\mathbf{A}\), with \(\nabla^2\mathbf{A} = -\mathbf{J}\), \(\mathbf{A}_t = 0\) and \(\partial A_n/\partial n = 0\) on the faces, so \(\nabla\times\mathbf{A}\) has no normal component and \(\mathbf{B}\) keeps that of \(\mathbf{B}_p\).
    5. Repeat 2–4 until \(\mathbf{B}\) stops changing.

    The P and N solutions use different halves of the α data. If the data were perfectly consistent with a force-free field, they would coincide; their difference measures that inconsistency (Wheatland & Régnier 2009).

    Internal units: \(\nabla\times\mathbf{B}\) in G/Mm, \(\alpha\) in Mm⁻¹ (\(\mu_0\) is absorbed in \(\mathbf{J}\)).

                   clip_percentile: float = 99.0) -> np.ndarray:
    """α_0 = J_z / B_z at the base (Mm⁻¹), 0 where |B_z| < ``threshold``·max|B_z|.

    Slightly smoothed (``smooth`` pixels) and the extreme values (noise where
    B_z is small) clipped at the ``clip_percentile`` percentile of |α|.
    """
    mag._require_vector()
    bx = gaussian_filter(mag.bx, smooth) if smooth else mag.bx
    by = gaussian_filter(mag.by, smooth) if smooth else mag.by
    bz = gaussian_filter(mag.bz, smooth) if smooth else mag.bz
    jz = np.gradient(by, mag.dx, axis=0, edge_order=2) - np.gradient(bx, mag.dx, axis=1, edge_order=2)
    strong = np.abs(mag.bz) > threshold * np.abs(mag.bz).max()
    a = np.zeros_like(bz)
    a[strong] = jz[strong] / bz[strong]
    if strong.any():
        lim = np.percentile(np.abs(a[strong]), clip_percentile)
        a = np.clip(a, -lim, lim)
    return a

    def nlfff_grad_rubin(mag: Magnetogram, nz: int, polarity: str = "P", initial: VectorField3D | None = None,
                     max_iter: int = 40, tol: float = 1e-3, relax: float = 0.5, stride: int | None = None,
                     alpha_threshold: float = 0.05, alpha_smooth: float = 1.0, ds: float = 0.5,
                     verbose: bool = True, return_result: bool = False):
    """NLFFF extrapolation by Grad–Rubin.

    **Parameters**

    - ``mag``: vector magnetogram (B_z is used as boundary condition and J_z/B_z
        on the chosen polarity).
    - ``nz``: levels in height.
    - ``polarity``: ``"P"`` (α of the positive polarity) or ``"N"`` (negative).
    - ``initial``: initial / reference field (by default the potential); its
        normal component on the six faces is kept.
    - ``max_iter``: Grad–Rubin iterations.
    - ``tol``: stop when the relative rms change of B between iterations < ``tol``.
    - ``relax``: under-relaxation B ← (1−r) B + r B_new (stabilises the iteration).
    - ``stride``: α is assigned by tracing from 1 of every ``stride`` points per axis and
        interpolated; 1 = every point (more accurate: in Low & Lou C_vec 0.980/0.987
        for P/N against 0.957/0.984 with 2). Default 1 up to ~2·10⁵ points and 2 above.
    """
    from .nlfff import NLFFFResult
    mag._require_vector()
    h = mag.dx
    pot = initial if initial is not None else potential_field(mag, nz)
    if stride is None:
        stride = 1 if mag.shape[0] * mag.shape[1] * nz <= 200_000 else 2
    bp = [pot.bx, pot.by, pot.bz]
    alpha0 = boundary_alpha(mag, alpha_threshold, alpha_smooth)
    pol = mag.bz > 0 if polarity == "P" else mag.bz < 0
    strong = pol & (np.abs(mag.bz) > alpha_threshold * np.abs(mag.bz).max())
    B = [c.copy() for c in bp]
    hist = []
    t0 = time.time()
    message, converged = "max_iter alcanzado", False
    for it in range(1, max_iter + 1):
        cur = VectorField3D(*B, dx=h, origin=pot.origin)
        a = _assign_alpha(cur, alpha0, polarity, strong, stride, ds)
        J = [a * B[i] for i in range(3)]
        # Cleaning: J ← J − ∇ψ, ∇²ψ = ∇·J, ∂ψ/∂n = 0 (∇·J = 0 without touching J_n)
        psi = poisson_neumann(divergence(*J, h), h)
        gpsi = np.gradient(psi, h, edge_order=2)
        J = [J[i] - gpsi[i] for i in range(3)]
        bj = current_field(*J, h)
        new = [bp[i] + bj[i] for i in range(3)]
        num = sum(((new[i] - B[i]) ** 2).sum() for i in range(3))
        den = sum((new[i] ** 2).sum() for i in range(3))
        change = float(np.sqrt(num / den))
        B = [(1 - relax) * B[i] + relax * new[i] for i in range(3)]
        e = float(sum((c ** 2).sum() for c in B))
        hist.append((it, change, e))
        if verbose:
            print(f"  Grad-Rubin ({polarity}) it {it:3d}  cambio rms {change:.2e}  "
                  f"E/E0 {e / float(sum((c ** 2).sum() for c in bp)):.4f}  ({time.time() - t0:.0f}s)", flush=True)
        if change < tol:
            converged, message = True, f"convergido: cambio rms = {change:.1e}"
            break
    out = VectorField3D(*B, dx=h, origin=pot.origin,
                        meta={"method": "NLFFF", "backend": f"grad-rubin-{polarity}", "polarity": polarity,
                              "iterations": it, "message": message, "nd": max(4, min(*mag.shape, nz) // 10),
                              "history": hist})
    if return_result:
        return NLFFFResult(out, [(i, c, np.nan) for i, c, _ in hist], converged, message)
    return out
