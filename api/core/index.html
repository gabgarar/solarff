    @dataclass
class Magnetogram:
    """Photospheric field on a local Cartesian grid.

    Attributes
    ----------
    bx, by, bz : ndarray (nx, ny) in Gauss. ``bz`` is the vertical
        (radial) component. ``bx``/``by`` may be ``None`` if only the vertical field is
        available (enough for the potential field and LFFF, not for NLFFF).
    dx : pixel size in Mm.
    origin : coordinates (x0, y0) in Mm of pixel [0, 0].
    meta : header / origin information.
    bx_err, by_err, bz_err : uncertainty (1σ, G) of each component at each
        pixel, if known (SHARP: segments Bp_err, Bt_err, Br_err). Used
        in the Monte Carlo (``solarff.monte_carlo``).
    """

    bz: np.ndarray
    bx: np.ndarray | None = None
    by: np.ndarray | None = None
    dx: float = 1.0
    meta: dict = field(default_factory=dict)
    origin: tuple = (0.0, 0.0)
    bx_err: np.ndarray | None = None
    by_err: np.ndarray | None = None
    bz_err: np.ndarray | None = None

    def __post_init__(self):
        self.bz = np.asarray(self.bz, dtype=float)
        if self.bx is not None:
            self.bx = np.asarray(self.bx, dtype=float)
            self.by = np.asarray(self.by, dtype=float)

    # ------------------------------------------------------------------ props
    @property
    def shape(self):
        return self.bz.shape

    @property
    def has_vector(self) -> bool:
        return self.bx is not None and self.by is not None

    @property
    def has_errors(self) -> bool:
        return self.bz_err is not None

    @property
    def bt(self):
        """Modulus of the transverse field."""
        self._require_vector()
        return np.hypot(self.bx, self.by)

    @property
    def x(self):
        return self.origin[0] + self.dx * np.arange(self.shape[0])

    @property
    def y(self):
        return self.origin[1] + self.dx * np.arange(self.shape[1])

    def _require_vector(self):
        if not self.has_vector:
            raise ValueError("Este magnetograma no tiene componentes transversales (bx, by).")

    # ------------------------------------------------------------- operations
    def _map(self, fn, dx=None, origin=None, efn=None):
        efn = efn or fn
        return Magnetogram(
            bz=fn(self.bz),
            bx=None if self.bx is None else fn(self.bx),
            by=None if self.by is None else fn(self.by),
            dx=self.dx if dx is None else dx,
            meta=dict(self.meta),
            origin=tuple(self.origin) if origin is None else tuple(origin),
            bx_err=None if self.bx_err is None else efn(self.bx_err),
            by_err=None if self.by_err is None else efn(self.by_err),
            bz_err=None if self.bz_err is None else efn(self.bz_err),
        )

    def crop(self, xslice: slice, yslice: slice) -> "Magnetogram":
        """Crop by pixel indices: ``mag.crop(slice(100, 400), slice(50, 250))``."""
        x0 = self.origin[0] + (xslice.start or 0) * self.dx
        y0 = self.origin[1] + (yslice.start or 0) * self.dx
        return self._map(lambda a: a[xslice, yslice].copy(), origin=(x0, y0))

    def rebin(self, factor: int) -> "Magnetogram":
        """Reduce the resolution by averaging ``factor x factor`` blocks.

        Essential with full-resolution SHARPs (~0.36 Mm/pixel): a 600x300x150
        NLFFF is not tractable in pure numpy.
        """
        if factor == 1:
            return self._map(lambda a: a.copy())
        nx, ny = (s // factor * factor for s in self.shape)

        def _bin(a):
            a = a[:nx, :ny]
            return a.reshape(nx // factor, factor, ny // factor, factor).mean(axis=(1, 3))

        def _bin_err(e):
            # Mean of f² pixels with independent errors: σ = sqrt(Σσ²) / f²
            return np.sqrt(_bin(e ** 2) / factor ** 2)

        off = (factor - 1) / 2 * self.dx
        return self._map(_bin, dx=self.dx * factor,
                         origin=(self.origin[0] + off, self.origin[1] + off), efn=_bin_err)

    def balance_flux(self) -> "Magnetogram":
        """Subtract the net flux of bz (needed for the problem to be well posed)."""
        out = self._map(lambda a: a.copy())
        out.bz = out.bz - out.bz.mean()
        return out

    def pad(self, n: int) -> "Magnetogram":
        """Add ``n`` pixels of zeros around (reduces edge effects)."""
        return self._map(lambda a: np.pad(a, n, mode="constant"), efn=lambda e: np.pad(e, n, mode="edge"),
                         origin=(self.origin[0] - n * self.dx, self.origin[1] - n * self.dx))

    # ----------------------------------------------------------- diagnostics
    def flux_imbalance(self) -> float:
        r"""Flux imbalance $|\sum B_z| / \sum |B_z|$."""
        return float(abs(self.bz.sum()) / np.abs(self.bz).sum())

    def jz(self) -> np.ndarray:
        """Vertical component of ∇×B at the photosphere (G/Mm)."""
        self._require_vector()
        return (np.gradient(self.by, self.dx, axis=0, edge_order=2)
                - np.gradient(self.bx, self.dx, axis=1, edge_order=2))

    def __repr__(self):
        kind = "vectorial" if self.has_vector else "sólo Bz"
        return (f"Magnetogram({kind}, shape={self.shape}, dx={self.dx:.4g} Mm, "
                f"|Bz|max={np.abs(self.bz).max():.0f} G)")

    def aly_criteria(mag: Magnetogram) -> dict:
    """Aly (1989) criteria: a photospheric field compatible with a force-free
    field must have zero net force and torque.

    Returns ``eps_force`` and ``eps_torque`` (dimensionless). Values << 1
    (typically < 0.1) indicate data consistent with NLFFF. Real
    photospheric magnetograms usually give high values, hence the
    preprocessing (``solarff.preprocess``).
    """
    mag._require_vector()
    bx, by, bz = mag.bx, mag.by, mag.bz
    nx, ny = mag.shape
    x, y = np.meshgrid(np.arange(nx) - (nx - 1) / 2, np.arange(ny) - (ny - 1) / 2, indexing="ij")
    b2 = bx ** 2 + by ** 2 + bz ** 2
    d = bz ** 2 - bx ** 2 - by ** 2
    eps_f = (abs((bx * bz).sum()) + abs((by * bz).sum()) + abs(d.sum())) / b2.sum()
    r = np.hypot(x, y)
    eps_t = (abs((x * d).sum()) + abs((y * d).sum()) + abs((y * bx * bz - x * by * bz).sum())) / (r * b2).sum()
    return {"eps_force": float(eps_f), "eps_torque": float(eps_t),
            "flux_imbalance": mag.flux_imbalance()}

class VectorField3D:
    """3D magnetic field on a uniform Cartesian grid.

    Attributes
    ----------
    bx, by, bz : ndarray (nx, ny, nz), in Gauss.
    dx : grid step (Mm), the same in the three directions.
    origin : coordinates (x0, y0, z0) of node [0, 0, 0] in Mm.
    meta : free dictionary (method, alpha, convergence history, ...).
    """

    bx: np.ndarray
    by: np.ndarray
    bz: np.ndarray
    dx: float
    origin: tuple = (0.0, 0.0, 0.0)
    meta: dict = field(default_factory=dict)

    @property
    def shape(self):
        return self.bx.shape

    @property
    def x(self):
        return self.origin[0] + self.dx * np.arange(self.shape[0])

    @property
    def y(self):
        return self.origin[1] + self.dx * np.arange(self.shape[1])

    @property
    def z(self):
        return self.origin[2] + self.dx * np.arange(self.shape[2])

    @property
    def extent(self):
        """(xmin, xmax, ymin, ymax, zmin, zmax) in Mm."""
        x, y, z = self.x, self.y, self.z
        return x[0], x[-1], y[0], y[-1], z[0], z[-1]

    @property
    def b2(self):
        return self.bx ** 2 + self.by ** 2 + self.bz ** 2

    @property
    def babs(self):
        return np.sqrt(self.b2)

    def current(self):
        """Return ∇×B (without the 4π/c factor; units G/Mm)."""
        return curl(self.bx, self.by, self.bz, self.dx)

    def divergence(self):
        return divergence(self.bx, self.by, self.bz, self.dx)

    def alpha(self, b_min_frac: float = 0.02) -> np.ndarray:
        r"""Local force-free parameter $\alpha = (\nabla\times\mathbf{B})\cdot\mathbf{B} / B^2$ (Mm⁻¹).

        NaN where |B| < ``b_min_frac``·max|B| (α is not well defined in
        weak field). In an LFFF it equals the constant α; in an NLFFF it is
        (ideally) constant along each field line.
        """
        jx, jy, jz = self.current()
        b2 = self.b2
        with np.errstate(invalid="ignore", divide="ignore"):
            a = (jx * self.bx + jy * self.by + jz * self.bz) / b2
        a[np.sqrt(b2) < b_min_frac * np.sqrt(b2.max())] = np.nan
        return a

    def bottom(self):
        """Components at z = 0 as a 2D tuple (bx, by, bz)."""
        return self.bx[:, :, 0], self.by[:, :, 0], self.bz[:, :, 0]

    def stack(self):
        return np.stack([self.bx, self.by, self.bz])

    def copy(self):
        return VectorField3D(self.bx.copy(), self.by.copy(), self.bz.copy(),
                             self.dx, tuple(self.origin), dict(self.meta))

    def __repr__(self):
        name = self.meta.get("method", "campo")
        return f"VectorField3D({name}, shape={self.shape}, dx={self.dx:.4g} Mm)"

    trapezoid_weights(shape) -> np.ndarray

    Trapezoid-rule weights on a grid of nodes (2D or 3D): 1/2 on each edge or face, 1/4 on edges and 1/8 at corners. Every volume or surface integral over the grid must use them: summing the nodes with weight 1 over-weights the edges, and at z = 0, where the field is strongest, biases the energy (revision R10).

    def trapezoid_weights(shape) -> np.ndarray:
    """Trapezoid-rule weights on a grid of nodes (2D or 3D): 1/2 on each edge or face,
    1/4 on edges and 1/8 at corners. Every volume or surface integral over the grid must
    use them: summing the nodes with weight 1 over-weights the edges, and at z = 0, where the field is
    strongest, biases the energy (revision R10)."""
    shape = tuple(shape)
    if shape not in _W:
        w = np.ones(shape)
        for ax in range(len(shape)):
            sl = [slice(None)] * len(shape)
            sl[ax] = 0
            w[tuple(sl)] *= 0.5
            sl[ax] = -1
            w[tuple(sl)] *= 0.5
        _W[shape] = w
    return _W[shape]

    def curl(bx, by, bz, h):
    """Curl of (bx, by, bz) on a uniform grid of step h."""
    dbx = np.gradient(bx, h, edge_order=2)
    dby = np.gradient(by, h, edge_order=2)
    dbz = np.gradient(bz, h, edge_order=2)
    jx = dbz[1] - dby[2]
    jy = dbx[2] - dbz[0]
    jz = dby[0] - dbx[1]
    return jx, jy, jz

    solarff.io ¶

    Reading and downloading of SDO/HMI data.

    Recommended product: SHARP CEA (hmi.sharp_cea_720s), segments

    • Br : radial component -> Bz
    • Bp : φ component (westward) -> Bx
    • Bt : θ component (southward) -> By = -Bt

    They are already reprojected onto a cylindrical equal-area (CEA) grid centred on the region, with the 180° ambiguity resolved and in heliographic components: exactly the local Cartesian boundary condition that the LFFF/NLFFF methods need.

    For potential/LFFF Br is enough (or the magnetogram segment of the hmi.sharp_cea_720s series, LOS, if the region is near disc centre).

    def load_sharp_cea(br: str, bp: str | None = None, bt: str | None = None,
                   rebin: int = 1) -> Magnetogram:
    """Load a SHARP CEA and convert it to the local Cartesian convention.

    **Parameters**

    - ``br, bp, bt``: paths to the FITS files of the Br, Bp, Bt segments. If only
        ``br`` is given, a Bz-only magnetogram is obtained (valid for potential and
        LFFF). A directory is also accepted in ``br``: the files
        ``*.Br.fits``, ``*.Bp.fits``, ``*.Bt.fits`` are looked for inside it.
    - ``rebin``: resolution reduction factor (see ``Magnetogram.rebin``).
    """
    if os.path.isdir(br):
        d = br
        found = {}
        for seg in ("Br", "Bp", "Bt", "Br_err", "Bp_err", "Bt_err"):
            m = sorted(glob.glob(os.path.join(d, f"*.{seg}.fits")))
            found[seg] = m[0] if m else None
        if found["Br"] is None:
            raise FileNotFoundError(f"No hay *.Br.fits en {d}")
        br, bp, bt = found["Br"], found["Bp"], found["Bt"]
        errs = (found["Br_err"], found["Bp_err"], found["Bt_err"])
    else:
        errs = (None, None, None)

    data_r, hdr = _read_fits(br)
    # FITS: data[row=y, column=x]  ->  our convention [ix, iy]
    bz = np.nan_to_num(data_r.T)
    bx = by = None
    if bp is not None and bt is not None:
        bx = np.nan_to_num(_read_fits(bp)[0].T)
        by = -np.nan_to_num(_read_fits(bt)[0].T)

    cdelt = float(hdr.get("CDELT1", 0.03))  # heliographic degrees per pixel
    rsun_m = float(hdr.get("RSUN_REF", 6.96e8))
    dx = np.deg2rad(cdelt) * rsun_m / 1e6  # Mm

    meta = {k: hdr.get(k) for k in ("T_REC", "HARPNUM", "NOAA_AR", "NOAA_ARS", "CRVAL1",
                                    "CRVAL2", "CRLN_OBS", "LON_FWT", "LAT_FWT", "USFLUX")
            if k in hdr}
    meta["source"] = SHARP_CEA
    lon = hdr.get("LON_FWT")
    if lon is not None and np.isfinite(lon) and abs(lon) > 45:
        warnings.warn(
            f"La región está a {lon:.0f}° del meridiano central: el ruido del campo "
            "transversal y los efectos de proyección son grandes. Preferible |lon| < 30-45°.")
    e = [None if f is None else np.nan_to_num(_read_fits(f)[0].T) for f in errs]
    mag = Magnetogram(bz=bz, bx=bx, by=by, dx=dx, meta=meta,
                      bz_err=e[0], bx_err=e[1], by_err=e[2])
    if mag.has_errors:
        meta["errors"] = "Br_err, Bp_err, Bt_err"
    return mag.rebin(rebin) if rebin > 1 else mag

    def from_sunpy(br, bp=None, bt=None, br_err=None, bp_err=None, bt_err=None,
               rebin: int = 1) -> Magnetogram:
    """Build a ``Magnetogram`` from sunpy Maps of SHARP CEA segments.

    Same conventions as ``load_sharp_cea``: ``Br -> Bz``, ``Bp -> Bx``,
    ``Bt -> -By``; ``Map.data`` is ``[y, x]`` and is transposed to ``[ix, iy]``;
    the pixel size in Mm comes from ``CDELT1`` (heliographic degrees) and
    ``RSUN_REF``. The full header of ``br`` is kept in ``meta["fits_header"]``
    so that ``to_sunpy`` can give it back.

    **Parameters**

    - ``br, bp, bt``: ``sunpy.map.Map`` of the ``Br``, ``Bp`` and ``Bt`` segments
        (``hmi.sharp_cea_720s``). With ``br`` alone the magnetogram has only ``Bz``.
    - ``br_err, bp_err, bt_err``: optional Maps of the ``*_err`` segments (1σ, G).
    - ``rebin``: resolution reduction factor (see ``Magnetogram.rebin``).
    """
    hdr = br.meta
    bz = np.nan_to_num(np.asarray(br.data, dtype=float).T)
    bx = by = None
    if bp is not None and bt is not None:
        bx = np.nan_to_num(np.asarray(bp.data, dtype=float).T)
        by = -np.nan_to_num(np.asarray(bt.data, dtype=float).T)

    cdelt = float(_header_get(hdr, "CDELT1", 0.03))
    rsun_m = float(_header_get(hdr, "RSUN_REF", 6.96e8))
    dx = np.deg2rad(cdelt) * rsun_m / 1e6  # Mm

    meta = {}
    for k in ("T_REC", "HARPNUM", "NOAA_AR", "NOAA_ARS", "CRVAL1", "CRVAL2", "CRLN_OBS",
              "LON_FWT", "LAT_FWT", "USFLUX"):
        v = _header_get(hdr, k)
        if v is not None:
            meta[k] = v
    meta["source"] = SHARP_CEA
    meta["fits_header"] = {str(k): v for k, v in dict(hdr).items()}

    def err(m):
        return None if m is None else np.nan_to_num(np.asarray(m.data, dtype=float).T)

    mag = Magnetogram(bz=bz, bx=bx, by=by, dx=dx, meta=meta,
                      bz_err=err(br_err), bx_err=err(bp_err), by_err=err(bt_err))
    if mag.has_errors:
        meta["errors"] = "Br_err, Bp_err, Bt_err"
    return mag.rebin(rebin) if rebin > 1 else mag

    def to_sunpy(mag: Magnetogram) -> dict:
    """Return the magnetogram as sunpy Maps: ``{"Br": Map, "Bp": Map, "Bt": Map}``.

    ``Bp`` and ``Bt`` are present only for a vector magnetogram. The header is the
    one stored by ``from_sunpy`` (``meta["fits_header"]``) when available;
    otherwise a minimal CEA header is built from ``dx`` and the keys kept in
    ``meta``. In both cases ``CDELT``, ``CRPIX`` and ``NAXIS`` are recomputed
    from the current arrays, so a rebinned magnetogram gives a consistent Map
    (the reference pixel is the centre of the map, as in SHARP CEA).
    """
    import sunpy.map

    nx, ny = mag.bz.shape
    rsun_m = float(mag.meta.get("RSUN_REF", 6.96e8))
    base = mag.meta.get("fits_header")
    if base is not None:
        rsun_m = float(_header_get(base, "RSUN_REF", rsun_m))
    cdelt = float(np.rad2deg(mag.dx * 1e6 / rsun_m))
    hdr = {}
    if base is not None:
        hdr.update({k: v for k, v in base.items()
                    if not str(k).upper().startswith(("NAXIS", "BITPIX", "CRPIX", "CDELT"))})
    else:
        hdr.update({"CTYPE1": "CRLN-CEA", "CTYPE2": "CRLT-CEA", "CUNIT1": "deg", "CUNIT2": "deg",
                    "CRVAL1": float(mag.meta.get("CRVAL1", 0.0)), "CRVAL2": float(mag.meta.get("CRVAL2", 0.0)),
                    "RSUN_REF": rsun_m, "TELESCOP": "SDO/HMI", "INSTRUME": "HMI_SIDE1"})
        t_rec = mag.meta.get("T_REC")
        if t_rec:  # "2011.02.15_00:00:00_TAI" -> ISO (TAI differs from UTC by ~37 s)
            d, t = str(t_rec).replace("_TAI", "").split("_")[:2]
            hdr["DATE-OBS"] = d.replace(".", "-") + "T" + t
        for k in ("HARPNUM", "NOAA_AR", "NOAA_ARS", "LON_FWT", "LAT_FWT", "USFLUX"):
            if k in mag.meta:
                hdr[k] = mag.meta[k]
    hdr.update({"CDELT1": cdelt, "CDELT2": cdelt, "CRPIX1": (nx + 1) / 2, "CRPIX2": (ny + 1) / 2,
                "CUNIT1": "deg", "CUNIT2": "deg"})

    def mk(a):
        return sunpy.map.Map(np.asarray(a, dtype=float).T.copy(), dict(hdr))

    out = {"Br": mk(mag.bz)}
    if mag.bx is not None and mag.by is not None:
        out["Bp"] = mk(mag.bx)
        out["Bt"] = mk(-mag.by)
    return out

    def search_sharp(time: str, noaa: int | None = None, email: str | None = None):
    """List the SHARPs available at a given time (requires ``drms`` and ``pandas``).

    Returns a DataFrame with HARPNUM, NOAA_ARS, position and total flux.

    """
    import drms

    c = drms.Client(email=email) if email else drms.Client()
    keys = c.query(f"{SHARP_CEA}[][{_jsoc_time(time)}]",
                   key="HARPNUM,NOAA_ARS,T_REC,LON_FWT,LAT_FWT,USFLUX")

    return keys

    def download_sharp(harpnum: int, time: str, email: str, out_dir: str = "data",
                   segments=("Br", "Bp", "Bt", "Br_err", "Bp_err", "Bt_err")) -> str:
    """Download the SHARP CEA segments from JSOC and return the directory.

    By default the error maps (``*_err``), needed for the Monte Carlo, are
    downloaded too. ``email`` must be registered at http://jsoc.stanford.edu/ajax/register_email.html.
    Then: ``load_sharp_cea(out_dir)``.
    """
    import drms

    os.makedirs(out_dir, exist_ok=True)
    c = drms.Client(email=email)
    ds = f"{SHARP_CEA}[{harpnum}][{_jsoc_time(time)}]{{{','.join(segments)}}}"
    req = c.export(ds, method="url", protocol="fits")
    req.wait()
    if req.status != 0:
        raise RuntimeError(f"Exportación JSOC fallida (status {req.status}) para {ds}")
    req.download(out_dir)
    return out_dir
